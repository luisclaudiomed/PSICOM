Este módulo avaliará o estilo de comunicação da pessoa com base em diversos fatores como estruturação da linguagem, tempo entre respostas, coesão lógica e variações de tom emocional.

def analyze_communication_complexity(self, conversation, profile_id):
    """
    Analisa a complexidade comunicacional com base em múltiplos fatores
    """
    if not conversation or not profile_id:
        return None
    
    # Filtrar mensagens do perfil alvo
    target_messages = [msg for msg in conversation if msg.get('is_target', False)]
    
    if not target_messages:
        return None
    
    # Análise de estruturação da linguagem
    structure_scores = []
    for msg in target_messages:
        # Verificar análise linguística se disponível
        if 'analysis' in msg and 'linguistic_analysis' in msg['analysis']:
            complexity = msg['analysis']['linguistic_analysis'].get('style_metrics', {}).get('complexity', 0)
            structure_scores.append(complexity)
    
    # Cálculo de tempo médio entre respostas
    response_times = []
    for i, msg in enumerate(conversation):
        if msg.get('is_target', False) and msg.get('context', {}).get('conversation_state', {}).get('is_response'):
            response_time = msg['context']['conversation_state'].get('response_time')
            if response_time is not None:
                response_times.append(response_time)
    
    # Análise de coesão lógica
    coherence_scores = []
    for msg in target_messages:
        # Calcular coesão lógica (uso de conectivos, estrutura de argumentação, etc.)
        text = msg.get('transcription') if msg.get('content_type') == 'audio' and msg.get('transcription') else msg.get('content', '')
        
        # Palavras que indicam coesão lógica
        coherence_markers = [
            'porque', 'portanto', 'assim', 'logo', 'então', 'dessa forma', 'por isso',
            'consequentemente', 'visto que', 'uma vez que', 'já que', 'desde que',
            'por conseguinte', 'por essa razão', 'desse modo', 'em vista disso'
        ]
        
        words = text.lower().split()
        # Calcular proporção de marcadores de coesão
        if words:
            coherence_score = sum(1 for word in words if any(marker in word for marker in coherence_markers)) / len(words)
            coherence_scores.append(coherence_score * 10)  # Normalizar para escala 0-10
    
    # Análise de variações de tom emocional
    emotion_variations = []
    emotions = [msg.get('analysis', {}).get('emotion', {}).get('primary_emotion') for msg in target_messages 
               if msg.get('analysis', {}).get('emotion', {}).get('primary_emotion')]
    
    if emotions:
        # Calcular quantas emoções diferentes são expressas
        unique_emotions = len(set(emotions))
        # Normalizar para escala 0-10
        emotion_variation = min(10, unique_emotions * 2)
        emotion_variations.append(emotion_variation)
    
    # Calcular pontuações médias
    avg_structure = sum(structure_scores) / len(structure_scores) if structure_scores else None
    avg_response_time = sum(response_times) / len(response_times) if response_times else None
    avg_coherence = sum(coherence_scores) / len(coherence_scores) if coherence_scores else None
    avg_emotion_variation = sum(emotion_variations) / len(emotion_variations) if emotion_variations else None
    
    # Determinar perfil de complexidade
    complexity_profile = {
        'structure_level': avg_structure,
        'response_time': avg_response_time,
        'coherence_level': avg_coherence,
        'emotional_variation': avg_emotion_variation,
    }
    
    # Calcular pontuação global de complexidade (0-10)
    components = [score for score in [avg_structure, avg_coherence, avg_emotion_variation] if score is not None]
    if components:
        global_complexity = sum(components) / len(components)
        complexity_profile['global_complexity'] = global_complexity
        
        # Classificar nível de complexidade
        if global_complexity < 3:
            complexity_profile['complexity_level'] = 'simples'
        elif global_complexity < 6:
            complexity_profile['complexity_level'] = 'moderado'
        else:
            complexity_profile['complexity_level'] = 'complexo'
    
    # Salvar no perfil para uso posterior
    if self.profiles and profile_id in self.profiles:
        if 'communication_patterns' not in self.profiles[profile_id]:
            self.profiles[profile_id]['communication_patterns'] = {}
        
        self.profiles[profile_id]['communication_patterns']['complexity_profile'] = complexity_profile
        self.update_profile(profile_id, self.profiles[profile_id])
    
    return complexity_profile

            "Vamos trabalhar nisso juntos.",

            "Sua perspectiva é válida."
        ],

            "Compreendo completamente sua posição.",

            "Sua preocupação é completamente justificável, vamos resolver isso."

def detect_behavioral_inconsistencies(self, message, profile_id):
    """
    Detecta incongruências entre padrões históricos de comunicação
    e comportamento recente, gerando perguntas reflexivas
    """
    if not message or not profile_id:
        return None
    
    # Obter o perfil
    profile = self.profiles.get(profile_id)
    if not profile:
        return None
    
    # Obter padrões comportamentais históricos
    historical_patterns = profile.get('behavioral_patterns', {})
    communication_style = historical_patterns.get('communication_style', {})
    
    # Obter análise comportamental da mensagem atual
    current_behavior = message.get('analysis', {}).get('behavioral_indicators', {})
    current_dominant = current_behavior.get('dominant_pattern', {}).get('pattern')
    
    # Se não houver análise comportamental ou padrões históricos, retornar
    if not current_dominant or not communication_style:
        return None
    
    # Identificar incongruências significativas
    inconsistencies = []
    
    # Verificar se o comportamento atual é diferente do padrão histórico
    if current_dominant in communication_style:
        historical_score = communication_style.get(current_dominant, 0)
        # Se o comportamento atual é raro historicamente (score < 0.2 ou 20%)
        if historical_score < 0.2:
            inconsistencies.append({
                'type': 'rare_behavior',
                'behavior': current_dominant,
                'historical_score': historical_score
            })
    
    # Verificar atitude atual vs. distribuição histórica
    current_attitude = message.get('analysis', {}).get('attitude', {}).get('primary_attitude')
    historical_attitudes = profile.get('communication_stats', {}).get('attitude_distribution', {})
    
    if current_attitude and historical_attitudes and current_attitude in historical_attitudes:
        historical_attitude_score = historical_attitudes.get(current_attitude, 0)
        # Se a atitude atual é rara historicamente (score < 0.2 ou 20%)
        if historical_attitude_score < 0.2:
            inconsistencies.append({
                'type': 'rare_attitude',
                'attitude': current_attitude,
                'historical_score': historical_attitude_score
            })
    
    # Verificar emoção atual vs. padrão emocional histórico
    current_emotion = message.get('analysis', {}).get('emotion', {}).get('primary_emotion')
    historical_emotions = profile.get('behavioral_patterns', {}).get('emotional_patterns', {})
    
    if current_emotion and historical_emotions and current_emotion in historical_emotions:
        historical_emotion_score = historical_emotions.get(current_emotion, 0)
        # Se a emoção atual é rara historicamente (score < 0.2 ou 20%)
        if historical_emotion_score < 0.2:
            inconsistencies.append({
                'type': 'rare_emotion',
                'emotion': current_emotion,
                'historical_score': historical_emotion_score
            })
    
    # Se não houver inconsistências significativas, retornar
    if not inconsistencies:
        return None
    
    # Gerar perguntas reflexivas baseadas nas inconsistências
    reflective_questions = []
    
    # Templates de perguntas para cada tipo de inconsistência
    behavior_questions = {
        'avoidance': [
            "O que te faz sentir que precisa ser cauteloso(a) neste assunto?",
            "Existe algo específico que está causando hesitação?"
        ],
        'approval_seeking': [
            "O que seria ideal para você nesta situação?",
            "O que te faria sentir mais confortável com esta decisão?"
        ],
        'control': [
            "O que te leva a querer mais estrutura neste momento?",
            "Existe alguma preocupação específica que motiva essa abordagem?"
        ],
        'emotional': [
            "Parece que este assunto é significativo para você. Gostaria de compartilhar mais sobre isso?",
            "O que faz este tema despertar esse tipo de resposta em você?"
        ],
        'proactive': [
            "O que motivou essa iniciativa?",
            "O que te inspirou a tomar essa abordagem?"
        ],
        'reactive': [
            "Como você se sente em relação a essa situação?",
            "O que seria um resultado ideal para você neste caso?"
        ]
    }
    
    # Templates para atitudes
    attitude_questions = {
        'evasiva': [
            "Existe algo que está gerando incerteza sobre este assunto?",
            "O que te ajudaria a sentir mais clareza nesta situação?"
        ],
        'submissa': [
            "O que você realmente preferiria neste caso?",
            "Se pudesse escolher livremente, qual seria sua opção?"
        ],
        'resistente': [
            "Existe alguma preocupação específica com essa proposta?",
            "O que seria necessário para tornar essa ideia mais aceitável?"
        ],
        'colaborativa': [
            "O que despertou esse interesse em colaborar neste assunto?",
            "Como podemos aproveitar melhor essa disposição para colaboração?"
        ]
    }
    
    # Gerar perguntas para cada inconsistência
    for inconsistency in inconsistencies:
        if inconsistency['type'] == 'rare_behavior':
            behavior = inconsistency['behavior']
            if behavior in behavior_questions:
                import random
                reflective_questions.append(random.choice(behavior_questions[behavior]))
        
        elif inconsistency['type'] == 'rare_attitude':
            attitude = inconsistency['attitude']
            if attitude in attitude_questions:
                import random
                reflective_questions.append(random.choice(attitude_questions[attitude]))
    
    # Limitar a 2 perguntas para não sobrecarregar
    reflective_questions = reflective_questions[:2]
    
    return {
        'inconsistencies': inconsistencies,
        'reflective_questions': reflective_questions
    }

def suggest_collaborative_engagement(self, message, context=None, profile_id=None):
    """
    Sugere frases que despertam senso de contribuição, responsabilidade 
    e envolvimento construtivo em situações de indecisão ou afastamento
    """
    if not message or not profile_id:
        return None
    
    # Obter o perfil
    profile = self.profiles.get(profile_id)
    if not profile:
        return None
    
    # Detectar sinais de indecisão ou afastamento
    indecision_signals = False
    disengagement_signals = False
    
    # Verificar comportamento
    behavior = message.get('analysis', {}).get('behavioral_indicators', {}).get('dominant_pattern', {}).get('pattern')
    if behavior in ['avoidance', 'reactive']:
        disengagement_signals = True
    
    # Verificar atitude
    attitude = message.get('analysis', {}).get('attitude', {}).get('primary_attitude')
    if attitude in ['evasiva', 'submissa']:
        indecision_signals = True
    
    # Analisar conteúdo da mensagem
    text = message.get('transcription') if message.get('content_type') == 'audio' and message.get('transcription') else message.get('content', '')
    
    # Marcadores de indecisão
    indecision_markers = [
        'não sei', 'talvez', 'pode ser', 'tanto faz', 'você decide', 'você escolhe',
        'qualquer um', 'como preferir', 'não tenho preferência', 'tanto faz',
        'você que sabe', 'não tenho opinião', 'como achar melhor'
    ]
    
    # Marcadores de afastamento
    disengagement_markers = [
        'depois vejo', 'mais tarde', 'quando puder', 'não tenho tempo',
        'estou ocupado', 'verei depois', 'agora não dá', 'não posso agora',
        'deixa para depois', 'em outro momento'
    ]
    
    if any(marker in text.lower() for marker in indecision_markers):
        indecision_signals = True
    
    if any(marker in text.lower() for marker in disengagement_markers):
        disengagement_signals = True
    
    # Se não houver sinais de indecisão ou afastamento, retornar
    if not indecision_signals and not disengagement_signals:
        return None
    
    # Gerar sugestões colaborativas
    collaborative_suggestions = []
    
    # Templates para indecisão
    indecision_templates = [
        "Sua perspectiva sobre {tema} seria muito valiosa para nossa decisão.",
        "Sua experiência traz um ângulo importante que enriqueceria esta discussão sobre {tema}.",
        "Acredito que você tem insights únicos sobre {tema} que poderiam nos ajudar.",
        "Sua contribuição seria fundamental para chegarmos a uma solução ideal para {tema}.",
        "Gostaria muito de considerar seu ponto de vista sobre {tema} antes de avançarmos."
    ]
    
    # Templates para afastamento
    disengagement_templates = [
        "Sua participação em {tema} faz uma diferença significativa no resultado final.",
        "O projeto de {tema} se beneficiaria muito com seu envolvimento, mesmo que breve.",
        "Sua colaboração prévia em {tema} trouxe resultados excelentes que gostaríamos de replicar.",
        "Percebo que quando você se envolve com {tema}, conseguimos avançar mais rapidamente.",
        "Seu conhecimento específico seria crucial para resolvermos este aspecto de {tema}."
    ]
    
    # Determinar tema da conversa
    tema = "este assunto"
    if message.get('analysis', {}).get('themes'):
        themes = message['analysis']['themes']
        if themes:
            # Usar o primeiro tema encontrado
            tema = themes[0].replace('#', '')
    
    # Verificar contexto anterior se disponível
    if context and 'previous_messages' in context:
        for prev_msg in context['previous_messages']:
            if 'themes' in prev_msg and prev_msg['themes']:
                tema = prev_msg['themes'][0].replace('#', '')
                break
    
    # Selecionar templates apropriados
    import random
    
    if indecision_signals:
        template = random.choice(indecision_templates)
        collaborative_suggestions.append(template.format(tema=tema))
    
    if disengagement_signals:
        template = random.choice(disengagement_templates)
        collaborative_suggestions.append(template.format(tema=tema))
    
    # Personalizar com base no perfil psicológico
    approval_need = profile.get('psychological_profile', {}).get('motivation_factors', {}).get('approval_need', {}).get('level')
    
    if approval_need == 'alta':
        appreciation_templates = [
            "Sempre valorizo muito suas contribuições em nossas discussões.",
            "Suas ideias anteriores foram muito úteis e apreciadas.",
            "Sua perspectiva única sempre agrega valor às nossas decisões."
        ]
        collaborative_suggestions.append(random.choice(appreciation_templates))
    
    return {
        'collaborative_suggestions': collaborative_suggestions,
        'signals': {
            'indecision': indecision_signals,
            'disengagement': disengagement_signals
        }
    }

def calibrate_affective_engagement(self, profile_id, recent_conversations=None, last_n_days=7):
    """
    Ajusta a intensidade e frequência de frases afetivas com base na 
    responsividade da pessoa ao longo do tempo
    """
    if not profile_id or profile_id not in self.profiles:
        return None
    
    profile = self.profiles[profile_id]
    
    # Se não houver conversas recentes fornecidas, buscar análises recentes
    if not recent_conversations:
        recent_conversations = []
        
        # Obter análises dos últimos dias
        if 'analyses' in profile:
            # Calcular data limite
            from datetime import datetime, timedelta
            cutoff_date = (datetime.now() - timedelta(days=last_n_days)).isoformat()
            
            # Filtrar análises recentes
            recent_analyses = [a for a in profile['analyses'] if a['timestamp'] >= cutoff_date]
            
            # Carregar conversas associadas a essas análises
            for analysis in recent_analyses:
                conversation_path = os.path.join(
                    self.config["DATA_DIR"], 
                    profile_id, 
                    "conversations", 
                    f"{analysis['id']}.json"
                )
                
                if os.path.exists(conversation_path):
                    try:
                        with open(conversation_path, 'r', encoding='utf-8') as f:
                            conv = json.load(f)
                            recent_conversations.extend(conv)
                    except Exception as e:
                        logger.error(f"Erro ao carregar conversa {analysis['id']}: {e}")
    
    # Se não houver conversas recentes, retornar com valores padrão
    if not recent_conversations:
        return {
            'affective_intensity': 'moderada',
            'affective_frequency': 0.3,  # 30% das mensagens
            'reason': 'default_values'
        }
    
    # Filtrar mensagens do perfil alvo
    target_messages = [msg for msg in recent_conversations if msg.get('is_target', False)]
    
    if not target_messages:
        return {
            'affective_intensity': 'moderada',
            'affective_frequency': 0.3,
            'reason': 'no_target_messages'
        }
    
    # Calcular métricas de responsividade
    
    # 1. Tempo médio de resposta
    response_times = []
    for msg in target_messages:
        if msg.get('context', {}).get('conversation_state', {}).get('is_response'):
            response_time = msg['context']['conversation_state'].get('response_time')
            if response_time is not None:
                response_times.append(response_time)
    
    avg_response_time = sum(response_times) / len(response_times) if response_times else None
    
    # 2. Taxa de resposta (quantas mensagens receberam resposta)
    # Isso requer uma análise mais detalhada da sequência de mensagens
    messages_requiring_response = 0
    messages_with_response = 0
    
    for i, msg in enumerate(recent_conversations):
        # Se a mensagem não é do alvo e não é a última da conversa
        if not msg.get('is_target', False) and i < len(recent_conversations) - 1:
            messages_requiring_response += 1
            
            # Verificar se a próxima mensagem é uma resposta do alvo
            next_msg = recent_conversations[i + 1]
            if next_msg.get('is_target', False):
                messages_with_response += 1
    
    response_rate = messages_with_response / messages_requiring_response if messages_requiring_response > 0 else 0
    
    # 3. Análise de atitudes e emoções nas respostas
    positive_attitudes = ['colaborativa']
    negative_attitudes = ['resistente', 'evasiva']
    
    attitude_counts = {'positive': 0, 'negative': 0, 'neutral': 0}
    
    for msg in target_messages:
        attitude = msg.get('analysis', {}).get('attitude', {}).get('primary_attitude')
        if attitude in positive_attitudes:
            attitude_counts['positive'] += 1
        elif attitude in negative_attitudes:
            attitude_counts['negative'] += 1
        else:
            attitude_counts['neutral'] += 1
    
    total_attitudes = sum(attitude_counts.values())
    positive_attitude_rate = attitude_counts['positive'] / total_attitudes if total_attitudes > 0 else 0
    negative_attitude_rate = attitude_counts['negative'] / total_attitudes if total_attitudes > 0 else 0
    
    # 4. Detectar padrões de abertura emocional
    emotion_openness = profile.get('psychological_profile', {}).get('emotional_tendencies', {}).get('emotional_openness', {}).get('level', 'média')
    
    # Calcular calibração de vínculo afetivo
    
    # Intensidade base com base na abertura emocional
    base_intensity = {
        'baixa': 'sutil',
        'média': 'moderada',
        'alta': 'alta'
    }.get(emotion_openness, 'moderada')
    
    # Ajustar com base na taxa de atitudes positivas/negativas
    if positive_attitude_rate > 0.6:  # Mais de 60% de atitudes positivas
        # Aumentar intensidade se base não for alta
        if base_intensity != 'alta':
            intensity = 'moderada' if base_intensity == 'sutil' else 'alta'
        else:
            intensity = base_intensity
    elif negative_attitude_rate > 0.6:  # Mais de 60% de atitudes negativas
        # Diminuir intensidade se base não for sutil
        if base_intensity != 'sutil':
            intensity = 'sutil' if base_intensity == 'alta' else 'moderada'
        else:
            intensity = base_intensity
    else:
        # Manter intensidade base
        intensity = base_intensity
    
    # Frequência de frases afetivas
    # Base: 30% das mensagens
    base_frequency = 0.3
    
    # Ajustar com base na taxa de resposta
    if response_rate > 0.8:  # Taxa de resposta alta (> 80%)
        frequency = base_frequency * 1.5  # Aumentar frequência
    elif response_rate < 0.5:  # Taxa de resposta baixa (< 50%)
        frequency = base_frequency * 0.7  # Diminuir frequência
    else:
        frequency = base_frequency
    
    # Limitar entre 10% e 80%
    frequency = max(0.1, min(0.8, frequency))
    
    # Construir resultado
    result = {
        'affective_intensity': intensity,
        'affective_frequency': frequency,
        'metrics': {
            'avg_response_time': avg_response_time,
            'response_rate': response_rate,
            'positive_attitude_rate': positive_attitude_rate,
            'negative_attitude_rate': negative_attitude_rate,
            'emotional_openness': emotion_openness
        },
        'reason': 'calibrated'
    }
    
    # Atualizar perfil para futuras referências
    if 'communication_patterns' not in profile:
        profile['communication_patterns'] = {}
    
    profile['communication_patterns']['affective_calibration'] = {
        'intensity': intensity,
        'frequency': frequency,
        'last_updated': datetime.now().isoformat()
    }
    
    self.update_profile(profile_id, profile)
    
    return result

    return {
        'recognition_phrases': recognition_phrases,

    }

def analyze_and_refine_communication(self, message, profile_id, conversation_context=None, previous_style_responses=None):
    """
    Função integradora que combina todos os módulos de comunicação para
    analisar a mensagem atual e oferecer recomendações de comunicação
    personalizadas e contextualmente apropriadas
    """
    if not message or not profile_id:
        return None
    
    # Verificar se o perfil existe
    if profile_id not in self.profiles:
        logger.error(f"Perfil não encontrado: {profile_id}")
        return None
    
    # Resultados de cada módulo
    results = {}
    
    # 1. Perfil de Complexidade Comunicacional
    if conversation_context and 'conversation' in conversation_context:
        complexity_profile = self.analyze_communication_complexity(
            conversation_context['conversation'], profile_id)
        results['complexity_profile'] = complexity_profile
    
    # 2. Personalização Afetiva Responsiva
    affective_responses = self.generate_affective_response(message, conversation_context, profile_id)
    results['affective_responses'] = affective_responses
    
    # 3. Estímulo de Consistência Comportamental
    behavioral_consistency = self.detect_behavioral_inconsistencies(message, profile_id)
    results['behavioral_consistency'] = behavioral_consistency
    
    # 4. Reforço de Participação Colaborativa
    collaborative_engagement = self.suggest_collaborative_engagement(message, conversation_context, profile_id)
    results['collaborative_engagement'] = collaborative_engagement
    
    # 5. Intensificação de Vínculo Baseado em Responsividade
    # Para este módulo, precisamos de dados ao longo do tempo
    if conversation_context and 'recent_conversations' in conversation_context:
        affective_calibration = self.calibrate_affective_engagement(
            profile_id, conversation_context['recent_conversations'])
        results['affective_calibration'] = affective_calibration
    
    # 6. Reconhecimento Direcionado por Mérito
    merit_recognition = self.generate_merit_based_recognition(message, profile_id)
    results['merit_recognition'] = merit_recognition
    
    # 7. Realce de Expectativas Construtivas por Referência Positiva
    positive_references = self.generate_positive_references(profile_id, conversation_context)
    results['positive_references'] = positive_references
    
    # 8. Reforço de Referências Positivas Anteriores
    positive_reinforcement = self.reinforce_positive_experiences(profile_id, conversation_context)
    results['positive_reinforcement'] = positive_reinforcement
    
    # 9. Alternância de Estilos de Comunicação Contextual
    communication_style = self.adapt_communication_style(message, profile_id, previous_style_responses)
    results['communication_style'] = communication_style
    
    # 10. Introduções com Sugestões Positivas
    positive_openers = self.generate_positive_openers(conversation_context, profile_id)
    results['positive_openers'] = positive_openers
    
    # Gerar recomendações consolidadas
    consolidated_recommendations = self.generate_consolidated_recommendations(results, profile_id)
    
    # Retornar resultados completos
    return {
        'module_results': results,
        'recommendations': consolidated_recommendations
    }

    )

        messagebox.showerror("Erro", "Conversa da análise não encontrada.")
        return
    
    try:
        # Carregar conversa
        with open(conversation_path, 'r', encoding='utf-8') as f:
            conversation = json.load(f)
        
        # Filtrar mensagens do perfil alvo
        target_messages = [msg for msg in conversation if msg.get('is_target', False)]

            messagebox.showinfo("Aviso", "Nenhuma mensagem do perfil alvo encontrada na conversa.")

        last_message = target_messages[-1]
        
        # Preparar contexto
        context = {
            'conversation': conversation,

def analyze_communication_complexity(self, conversation, profile_id):
    """Análise da complexidade comunicacional com base em múltiplos fatores"""
    if not conversation or not profile_id:
        return None
    
    target_messages = [msg for msg in conversation if msg.get('is_target', False)]
    
    # Análise de estruturação linguística
    structure_scores = []
    for msg in target_messages:
        if 'analysis' in msg and 'linguistic_analysis' in msg['analysis']:
            complexity = msg['analysis']['linguistic_analysis'].get('style_metrics', {}).get('complexity', 0)
            structure_scores.append(complexity)
    
    # Tempo médio entre respostas
    response_times = []
    for msg in target_messages:
        if msg.get('context', {}).get('conversation_state', {}).get('is_response'):
            response_time = msg['context']['conversation_state'].get('response_time')
            if response_time is not None:
                response_times.append(response_time)
    
    # Análise de coesão lógica
    coherence_scores = []
    for msg in target_messages:
        text = msg.get('transcription') if msg.get('content_type') == 'audio' and msg.get('transcription') else msg.get('content', '')
        
        coherence_markers = [
            'porque', 'portanto', 'assim', 'logo', 'então', 'dessa forma', 'por isso',
            'consequentemente', 'visto que', 'uma vez que', 'já que', 'desde que'
        ]
        
        words = text.lower().split()
        if words:
            coherence_score = sum(1 for word in words if any(marker in word for marker in coherence_markers)) / len(words)
            coherence_scores.append(coherence_score * 10)  # Normalizar para escala 0-10
    
    # Análise de variações de tom emocional
    emotion_variations = []
    emotions = [msg.get('analysis', {}).get('emotion', {}).get('primary_emotion') for msg in target_messages 
               if msg.get('analysis', {}).get('emotion', {}).get('primary_emotion')]
    
    if emotions:
        unique_emotions = len(set(emotions))
        emotion_variation = min(10, unique_emotions * 2)
        emotion_variations.append(emotion_variation)
    
    # Cálculos finais
    avg_structure = sum(structure_scores) / len(structure_scores) if structure_scores else None
    avg_response_time = sum(response_times) / len(response_times) if response_times else None
    avg_coherence = sum(coherence_scores) / len(coherence_scores) if coherence_scores else None
    avg_emotion_variation = sum(emotion_variations) / len(emotion_variations) if emotion_variations else None
    
    complexity_profile = {
        'structure_level': avg_structure,
        'response_time': avg_response_time,
        'coherence_level': avg_coherence,
        'emotional_variation': avg_emotion_variation,
    }
    
    # Complexidade global
    components = [score for score in [avg_structure, avg_coherence, avg_emotion_variation] if score is not None]
    if components:
        global_complexity = sum(components) / len(components)
        complexity_profile['global_complexity'] = global_complexity
        
        if global_complexity < 3:
            complexity_profile['complexity_level'] = 'simples'
        elif global_complexity < 6:
            complexity_profile['complexity_level'] = 'moderado'
        else:
            complexity_profile['complexity_level'] = 'complexo'
    
    # Salvamento no perfil
    if self.profiles and profile_id in self.profiles:
        if 'communication_patterns' not in self.profiles[profile_id]:
            self.profiles[profile_id]['communication_patterns'] = {}
        
        self.profiles[profile_id]['communication_patterns']['complexity_profile'] = complexity_profile
        self.update_profile(profile_id, self.profiles[profile_id])
    
    return complexity_profile

def generate_affective_response(self, message, context=None, profile_id=None):
    """Gera respostas com tom afetivo ao detectar insegurança ou necessidade de acolhimento"""
    if not message or not profile_id:
        return None
    
    profile = self.profiles.get(profile_id)
    if not profile:
        return None
    
    # Analisar indicadores de insegurança
    text = message.get('transcription') if message.get('content_type') == 'audio' and message.get('transcription') else message.get('content', '')
    
    insecurity_markers = [
        'não sei', 'talvez', 'acho que', 'será que', 'não tenho certeza', 
        'preciso de ajuda', 'estou preocupado', 'estou ansioso', 'receio',
        'desculpe', 'perdão', 'não estou bem', 'está difícil'
    ]
    
    # Verificar análises existentes
    emotion_indicators = False
    if (message.get('analysis', {}).get('emotion', {}).get('primary_emotion') in 
        ['hesitação', 'tensão', 'ansiedade', 'tristeza']):
        emotion_indicators = True
    
    attitude_indicators = False
    if message.get('analysis', {}).get('attitude', {}).get('primary_attitude') in ['evasiva', 'submissa']:
        attitude_indicators = True
    
    psych_indicators = False
    vulnerability = message.get('analysis', {}).get('psychological_indicators', {}).get('vulnerability', {}).get('level')
    if vulnerability in ['média', 'alta']:
        psych_indicators = True

def detect_behavioral_inconsistencies(self, message, profile_id):
    """Detecta incongruências entre padrões comunicacionais históricos e comportamento recente"""
    if not message or not profile_id:
        return None
    
    profile = self.profiles.get(profile_id)
    if not profile:
        return None
    
    historical_patterns = profile.get('behavioral_patterns', {})
    communication_style = historical_patterns.get('communication_style', {})
    
    current_behavior = message.get('analysis', {}).get('behavioral_indicators', {})
    current_dominant = current_behavior.get('dominant_pattern', {}).get('pattern')
    
    if not current_dominant or not communication_style:
        return None
    
    # Identificar incongruências significativas
    inconsistencies = []
    
    # Comportamento atual vs. histórico
    if current_dominant in communication_style:
        historical_score = communication_style.get(current_dominant, 0)
        if historical_score < 0.2:  # Comportamento raro (menos de 20% do histórico)
            inconsistencies.append({
                'type': 'rare_behavior',
                'behavior': current_dominant,
                'historical_score': historical_score
            })
    
    # Atitude atual vs. histórico
    current_attitude = message.get('analysis', {}).get('attitude', {}).get('primary_attitude')
    historical_attitudes = profile.get('communication_stats', {}).get('attitude_distribution', {})
    
    if current_attitude and historical_attitudes and current_attitude in historical_attitudes:
        historical_attitude_score = historical_attitudes.get(current_attitude, 0)
        if historical_attitude_score < 0.2:
            inconsistencies.append({
                'type': 'rare_attitude',
                'attitude': current_attitude,
                'historical_score': historical_attitude_score
            })
    
    if not inconsistencies:
        return None
    
    # Gerar perguntas reflexivas baseadas nas inconsistências
    reflective_questions = []
    
    # Templates de perguntas para cada tipo de comportamento
    behavior_questions = {
        'avoidance': [
            "O que te faz sentir que precisa ser cauteloso(a) neste assunto?",
            "Existe algo específico que está causando hesitação?"
        ],
        'approval_seeking': [
            "O que seria ideal para você nesta situação?",
            "O que te faria sentir mais confortável com esta decisão?"
        ],
        'control': [
            "O que te leva a querer mais estrutura neste momento?",
            "Existe alguma preocupação específica que motiva essa abordagem?"
        ],
        'emotional': [
            "Parece que este assunto é significativo para você. Gostaria de compartilhar mais sobre isso?",
            "O que faz este tema despertar esse tipo de resposta em você?"
        ]
    }
    
    # Templates para atitudes
    attitude_questions = {
        'evasiva': [
            "Existe algo que está gerando incerteza sobre este assunto?",
            "O que te ajudaria a sentir mais clareza nesta situação?"
        ],
        'submissa': [
            "O que você realmente preferiria neste caso?",
            "Se pudesse escolher livremente, qual seria sua opção?"
        ],
        'resistente': [
            "Existe alguma preocupação específica com essa proposta?",
            "O que seria necessário para tornar essa ideia mais aceitável?"
        ]
    }
    
    # Gerar perguntas para cada inconsistência
    import random
    for inconsistency in inconsistencies:
        if inconsistency['type'] == 'rare_behavior':
            behavior = inconsistency['behavior']
            if behavior in behavior_questions:
                reflective_questions.append(random.choice(behavior_questions[behavior]))
        
        elif inconsistency['type'] == 'rare_attitude':
            attitude = inconsistency['attitude']
            if attitude in attitude_questions:
                reflective_questions.append(random.choice(attitude_questions[attitude]))
    
    # Limitar a 2 perguntas
    reflective_questions = reflective_questions[:2]
    
    return {
        'inconsistencies': inconsistencies,
        'reflective_questions': reflective_questions
    }

def suggest_collaborative_engagement(self, message, context=None, profile_id=None):

    if not profile:
        return None

    indecision_signals = False
    disengagement_signals = False

    behavior = message.get('analysis', {}).get('behavioral_indicators', {}).get('dominant_pattern', {}).get('pattern')
    if behavior in ['avoidance', 'reactive']:
        disengagement_signals = True
    
    attitude = message.get('analysis', {}).get('attitude', {}).get('primary_attitude')
    if attitude in ['evasiva', 'submissa']:
        indecision_signals = True
    
    # Analisar conteúdo da mensagem
    text = message.get('transcription') if message.get('content_type') == 'audio' and message.get('transcription') else message.get('content', '')
    
    # Marcadores linguísticos
    indecision_markers = [
        'não sei', 'talvez', 'pode ser', 'tanto faz', 'você decide', 'você escolhe',
        'qualquer um', 'como preferir', 'não tenho preferência'
    ]
    
    disengagement_markers = [
        'depois vejo', 'mais tarde', 'quando puder', 'não tenho tempo',
        'estou ocupado', 'verei depois', 'agora não dá'
    ]
    
    if any(marker in text.lower() for marker in indecision_markers):
        indecision_signals = True
    
    if any(marker in text.lower() for marker in disengagement_markers):
        disengagement_signals = True
    
    if not indecision_signals and not disengagement_signals:
        return None
    
    # Templates para indecisão
    indecision_templates = [
        "Sua perspectiva sobre {tema} seria muito valiosa para nossa decisão.",
        "Sua experiência traz um ângulo importante que enriqueceria esta discussão sobre {tema}.",

        "Sua contribuição seria fundamental para chegarmos a uma solução ideal para {tema}."
    ]
    
    # Templates para afastamento
    disengagement_templates = [

        "Percebo que quando você se envolve com {tema}, conseguimos avançar mais rapidamente."
    ]
    
    # Determinar tema da conversa
    tema = "este assunto"
    if message.get('analysis', {}).get('themes'):
        themes = message['analysis']['themes']
        if themes:
            tema = themes[0].replace('#', '')
    
    # Verificar contexto anterior
    if context and 'previous_messages' in context:
        for prev_msg in context['previous_messages']:
            if 'themes' in prev_msg and prev_msg['themes']:
                tema = prev_msg['themes'][0].replace('#', '')
                break
    
    # Selecionar templates apropriados
    import random
    collaborative_suggestions = []
    
    if indecision_signals:
        template = random.choice(indecision_templates)
        collaborative_suggestions.append(template.format(tema=tema))
    
    if disengagement_signals:
        template = random.choice(disengagement_templates)
        collaborative_suggestions.append(template.format(tema=tema))

    approval_need = profile.get('psychological_profile', {}).get('motivation_factors', {}).get('approval_need', {}).get('level')

    return {
        'collaborative_suggestions': collaborative_suggestions,

        }

def calibrate_affective_engagement(self, profile_id, recent_conversations=None, last_n_days=7):
    """Ajusta a intensidade e frequência de frases afetivas com base na responsividade"""
    if not profile_id or profile_id not in self.profiles:
        return None
    
    profile = self.profiles[profile_id]
    
    # Se não houver conversas recentes, buscar análises recentes
    if not recent_conversations:
        recent_conversations = []
        
        if 'analyses' in profile:
            # Calcular data limite
            from datetime import datetime, timedelta
            cutoff_date = (datetime.now() - timedelta(days=last_n_days)).isoformat()
            
            # Filtrar análises recentes
            recent_analyses = [a for a in profile['analyses'] if a['timestamp'] >= cutoff_date]
            
            # Carregar conversas
            for analysis in recent_analyses:
                conversation_path = os.path.join(
                    self.config["DATA_DIR"], 
                    profile_id, 
                    "conversations", 
                    f"{analysis['id']}.json"
                )
                
                if os.path.exists(conversation_path):
                    try:
                        with open(conversation_path, 'r', encoding='utf-8') as f:
                            conv = json.load(f)
                            recent_conversations.extend(conv)
                    except Exception as e:
                        logger.error(f"Erro ao carregar conversa {analysis['id']}: {e}")
    
    # Valores padrão se não houver dados suficientes
    if not recent_conversations:
        return {
            'affective_intensity': 'moderada',
            'affective_frequency': 0.3,  # 30% das mensagens
            'reason': 'default_values'
        }
    
    # Filtrar mensagens do perfil alvo
    target_messages = [msg for msg in recent_conversations if msg.get('is_target', False)]
    
    if not target_messages:
        return {
            'affective_intensity': 'moderada',
            'affective_frequency': 0.3,
            'reason': 'no_target_messages'
        }
    
    # Cálculo de métricas de responsividade
    
    # 1. Tempo médio de resposta
    response_times = []
    for msg in target_messages:
        if msg.get('context', {}).get('conversation_state', {}).get('is_response'):
            response_time = msg['context']['conversation_state'].get('response_time')
            if response_time is not None:
                response_times.append(response_time)
    
    avg_response_time = sum(response_times) / len(response_times) if response_times else None
    
    # 2. Taxa de resposta
    messages_requiring_response = 0
    messages_with_response = 0
    
    for i, msg in enumerate(recent_conversations):
        if not msg.get('is_target', False) and i < len(recent_conversations) - 1:
            messages_requiring_response += 1
            
            # Verificar se a próxima mensagem é uma resposta do alvo
            next_msg = recent_conversations[i + 1]
            if next_msg.get('is_target', False):
                messages_with_response += 1
    
    response_rate = messages_with_response / messages_requiring_response if messages_requiring_response > 0 else 0
    
    # 3. Análise de atitudes
    positive_attitudes = ['colaborativa']
    negative_attitudes = ['resistente', 'evasiva']
    
    attitude_counts = {'positive': 0, 'negative': 0, 'neutral': 0}
    
    for msg in target_messages:
        attitude = msg.get('analysis', {}).get('attitude', {}).get('primary_attitude')
        if attitude in positive_attitudes:
            attitude_counts['positive'] += 1
        elif attitude in negative_attitudes:
            attitude_counts['negative'] += 1
        else:
            attitude_counts['neutral'] += 1
    
    total_attitudes = sum(attitude_counts.values())
    positive_attitude_rate = attitude_counts['positive'] / total_attitudes if total_attitudes > 0 else 0
    negative_attitude_rate = attitude_counts['negative'] / total_attitudes if total_attitudes > 0 else 0
    
    # 4. Verificar abertura emocional
    emotion_openness = profile.get('psychological_profile', {}).get('emotional_tendencies', {}).get('emotional_openness', {}).get('level', 'média')
    
    # Calibração da intensidade afetiva
    base_intensity = {
        'baixa': 'sutil',
        'média': 'moderada',
        'alta': 'alta'
    }.get(emotion_openness, 'moderada')
    
    # Ajustar com base nas atitudes
    if positive_attitude_rate > 0.6:  # Mais de 60% positivas
        intensity = 'moderada' if base_intensity == 'sutil' else 'alta'
    elif negative_attitude_rate > 0.6:  # Mais de 60% negativas
        intensity = 'sutil' if base_intensity == 'alta' else 'moderada'
    else:
        intensity = base_intensity
    
    # Frequência de frases afetivas (base: 30%)
    base_frequency = 0.3
    
    # Ajustar com base na taxa de resposta
    if response_rate > 0.8:  # Taxa de resposta alta
        frequency = base_frequency * 1.5  # Aumentar
    elif response_rate < 0.5:  # Taxa de resposta baixa
        frequency = base_frequency * 0.7  # Diminuir
    else:
        frequency = base_frequency
    
    # Limitar entre 10% e 80%
    frequency = max(0.1, min(0.8, frequency))
    
    # Resultado final
    result = {
        'affective_intensity': intensity,
        'affective_frequency': frequency,
        'metrics': {
            'avg_response_time': avg_response_time,
            'response_rate': response_rate,
            'positive_attitude_rate': positive_attitude_rate,
            'negative_attitude_rate': negative_attitude_rate,
            'emotional_openness': emotion_openness
        },
        'reason': 'calibrated'
    }
    
    # Atualizar perfil
    if 'communication_patterns' not in profile:
        profile['communication_patterns'] = {}
    
    profile['communication_patterns']['affective_calibration'] = {
        'intensity': intensity,
        'frequency': frequency,
        'last_updated': datetime.now().isoformat()
    }
    
    self.update_profile(profile_id, profile)
    
    return result

def reinforce_positive_experiences(self, profile_id, current_context):
    """Identifica e reforça experiências positivas anteriores para consolidar padrões positivos"""
    if not profile_id or not current_context:
        return None
    
    profile = self.profiles.get(profile_id)
    if not profile:
        return None
    
    # Extrair tema atual
    current_theme = None
    if 'themes' in current_context and current_context['themes']:
        current_theme = current_context['themes'][0].replace('#', '')
    
    # Buscar interações anteriores positivas
    positive_experiences = []
    
    if 'analyses' in profile:
        # Até 10 análises mais recentes, em ordem reversa
        for analysis in sorted(profile['analyses'], key=lambda x: x['timestamp'], reverse=True)[:10]:
            # Verificar recomendações da análise com menções positivas
            if 'summary' in analysis and 'recommendations' in analysis['summary']:
                for rec in analysis['summary']['recommendations'].get('general', []):
                    if any(term in rec.lower() for term in ['sucesso', 'eficaz', 'positiv', 'bem-sucedid', 'funcionou bem']):
                        positive_experiences.append({
                            'type': 'recommendation',
                            'content': rec,
                            'timestamp': analysis['timestamp'],
                            'reference': f"análise de {datetime.fromisoformat(analysis['timestamp']).strftime('%d/%m/%Y')}"
                        })
            
            # Buscar interações específicas positivas
            conversation_path = os.path.join(
                self.config["DATA_DIR"], 
                profile_id, 
                "conversations", 
                f"{analysis['id']}.json"
            )
            
            if not os.path.exists(conversation_path):
                continue
            
            try:
                with open(conversation_path, 'r', encoding='utf-8') as f:
                    conversation = json.load(f)
                
                # Buscar reconhecimentos positivos
                for i, msg in enumerate(conversation):
                    # Se não é mensagem do alvo e contém termos de reconhecimento
                    if not msg.get('is_target', False):
                        text = msg.get('content', '').lower()
                        
                        if any(term in text for term in ['parabéns', 'excelente', 'ótimo trabalho', 'muito bom', 'agradeço']):
                            # Verificar se há resposta do alvo
                            if i < len(conversation) - 1 and conversation[i+1].get('is_target', False):
                                next_msg = conversation[i+1]
                                
                                # Extrair tema se disponível
                                msg_theme = None
                                if 'analysis' in next_msg and 'themes' in next_msg['analysis'] and next_msg['analysis']['themes']:
                                    msg_theme = next_msg['analysis']['themes'][0].replace('#', '')
                                
                                # Só adicionar se compatível com tema atual
                                if not current_theme or not msg_theme or current_theme == msg_theme:
                                    positive_experiences.append({
                                        'type': 'interaction',
                                        'content': f"reconhecimento recebido em conversa sobre {msg_theme or 'assunto anterior'}",
                                        'context': msg.get('content'),
                                        'response': next_msg.get('content') or next_msg.get('transcription', ''),
                                        'timestamp': next_msg.get('datetime'),
                                        'theme': msg_theme
                                    })
            
            except Exception as e:
                logger.error(f"Erro ao carregar conversa {analysis['id']}: {e}")
    
    # Se não houver experiências específicas, usar referências gerais
    if not positive_experiences:
        behavioral_patterns = profile.get('behavioral_patterns', {})
        communication_style = behavioral_patterns.get('communication_style', {})
        
        if communication_style:
            if communication_style.get('proactive', 0) > 0.3 or communication_style.get('collaborative', 0) > 0.3:
                positive_experiences.append({
                    'type': 'profile_pattern',
                    'content': "tendência para abordagens proativas e colaborativas",
                    'behavior': 'proactive' if communication_style.get('proactive', 0) > communication_style.get('collaborative', 0) else 'collaborative'
                })
            
        # Verificar atitudes no histórico
        attitude_dist = profile.get('communication_stats', {}).get('attitude_distribution', {})
        if attitude_dist and attitude_dist.get('colaborativa', 0) > 0.3:
            positive_experiences.append({
                'type': 'profile_pattern',
                'content': "histórico de atitudes colaborativas",
                'attitude': 'colaborativa'
            })
    
    # Se ainda não houver experiências positivas, não podemos fornecer reforço
    if not positive_experiences:
        return None
    
    # Priorizar experiências relacionadas ao tema atual
    themed_experiences = [exp for exp in positive_experiences if exp.get('theme') == current_theme]
    
    if themed_experiences:
        selected_experiences = themed_experiences[:2]
    else:
        # Ordenar por recência
        sorted_experiences = sorted(positive_experiences, key=lambda x: x.get('timestamp', ''), reverse=True)
        selected_experiences = sorted_experiences[:2]
    
    # Templates por tipo de experiência
    import random
    reinforcement_phrases = []
    
    recommendation_templates = [
        "Como observamos anteriormente, {content}",
        "Retomando o que analisamos antes, {content}",
        "Em sintonia com o que já havíamos identificado, {content}"
    ]
    
    interaction_templates = [
        "Em situação anterior, sua abordagem para {theme} recebeu reconhecimento positivo.",
        "Como demonstrado anteriormente em conversa sobre {theme}, sua contribuição foi muito bem recebida.",
        "Retomando um ponto forte seu: em discussão anterior sobre {theme}, sua perspectiva trouxe valor significativo."
    ]
    
    profile_templates = [
        "Seu histórico de {content} é um diferencial importante que podemos aproveitar neste momento.",
        "Sua tendência natural para {content} pode ser particularmente valiosa nesta situação.",
        "Algo que tenho observado consistentemente é sua capacidade para {content}, que seria muito útil agora."
    ]
    
    for experience in selected_experiences:
        exp_type = experience['type']
        
        if exp_type == 'recommendation':
            template = random.choice(recommendation_templates)
            content = experience['content'].lower()
            # Reescrever para terceira pessoa se estiver em segunda
            content = content.replace("você deve", "seria benéfico")
            content = content.replace("use ", "usar ")
            reinforcement_phrases.append(template.format(content=content))
            
        elif exp_type == 'interaction':
            template = random.choice(interaction_templates)
            theme = experience.get('theme', 'um assunto importante')
            reinforcement_phrases.append(template.format(theme=theme))
            
        elif exp_type == 'profile_pattern':
            template = random.choice(profile_templates)
            content = experience['content']
            reinforcement_phrases.append(template.format(content=content))
    
    return {
        'reinforcement_phrases': reinforcement_phrases,
        'experiences_found': len(positive_experiences)
    }

def analyze_and_refine_communication(self, message, profile_id, conversation_context=None, previous_style_responses=None):
    """
    Função integradora que combina todos os módulos para análise completa e recomendações
    """
    if not message or not profile_id or profile_id not in self.profiles:
        return None
    
    # Resultados de cada módulo
    results = {}
    
    # 1. Perfil de Complexidade Comunicacional
    if conversation_context and 'conversation' in conversation_context:
        results['complexity_profile'] = self.analyze_communication_complexity(
            conversation_context['conversation'], profile_id)
    
    # 2. Personalização Afetiva Responsiva
    results['affective_responses'] = self.generate_affective_response(
        message, conversation_context, profile_id)
    
    # 3. Estímulo de Consistência Comportamental
    results['behavioral_consistency'] = self.detect_behavioral_inconsistencies(
        message, profile_id)
    
    # 4. Reforço de Participação Colaborativa
    results['collaborative_engagement'] = self.suggest_collaborative_engagement(
        message, conversation_context, profile_id)
    
    # 5. Intensificação de Vínculo
    if conversation_context and 'recent_conversations' in conversation_context:
        results['affective_calibration'] = self.calibrate_affective_engagement(
            profile_id, conversation_context['recent_conversations'])
    
    # 6. Reconhecimento Direcionado por Mérito
    results['merit_recognition'] = self.generate_merit_based_recognition(
        message, profile_id)
    
    # 7. Realce de Expectativas por Referência
    results['positive_references'] = self.generate_positive_references(
        profile_id, conversation_context)
    
    # 8. Reforço de Referências Positivas
    results['positive_reinforcement'] = self.reinforce_positive_experiences(
        profile_id, conversation_context)
    
    # 9. Alternância de Estilos de Comunicação
    results['communication_style'] = self.adapt_communication_style(
        message, profile_id, previous_style_responses)
    
    # 10. Introduções com Sugestões Positivas
    results['positive_openers'] = self.generate_positive_openers(
        conversation_context, profile_id)
    
    # Gerar recomendações consolidadas
    consolidated_recommendations = self.generate_consolidated_recommendations(
        results, profile_id)
    
    return {
        'module_results': results,
        'recommendations': consolidated_recommendations
    }
