12. Modulação de Resposta por Variação Emocional Detectada

Este módulo analisa mudanças emocionais ou de responsividade entre mensagens e sugere ajustes no tom da próxima resposta.

    def modulate_response_by_emotion_variation(self, current_message, previous_message, profile_id):
        """
        Ajusta o tom da resposta com base em variações emocionais ou de
        responsividade detectadas entre a mensagem atual e a anterior.
        """
        if not current_message or not previous_message or not profile_id:
            return {'suggested_tone_adjustment': 'maintain_neutrality', 'reason': 'missing_data'}

        profile = self.profiles.get(profile_id)
        if not profile:
            return {'suggested_tone_adjustment': 'maintain_neutrality', 'reason': 'profile_not_found'}

Comparar estado emocional
        current_emotion_data = current_message.get('analysis', {}).get('emotion', {})
        previous_emotion_data = previous_message.get('analysis', {}).get('emotion', {})

        emotion_shift = 'stable'
        if current_emotion_data and previous_emotion_data:
            current_valence = current_emotion_data.get('valence', 5) # Escala 0-10
            previous_valence = previous_emotion_data.get('valence', 5)
            current_arousal = current_emotion_data.get('arousal', 5)
            previous_arousal = previous_emotion_data.get('arousal', 5)

            valence_diff = current_valence - previous_valence
            arousal_diff = current_arousal - previous_arousal

            if abs(valence_diff) > 2: # Mudança significativa na valência (>2 pontos)
                emotion_shift = 'more_positive' if valence_diff > 0 else 'more_negative'
            elif abs(arousal_diff) > 2: # Mudança significativa na ativação
                 emotion_shift = 'more_aroused' if arousal_diff > 0 else 'less_aroused'

Comparar tempo de resposta (se aplicável)
        response_time_shift = 'stable'
        current_response_time = current_message.get('context', {}).get('conversation_state', {}).get('response_time')
        previous_response_time = previous_message.get('context', {}).get('conversation_state', {}).get('response_time')
        avg_historical_time = profile.get('communication_stats', {}).get('avg_response_time')

        if current_response_time is not None:
             comparison_time = previous_response_time if previous_response_time is not None else avg_historical_time
             if comparison_time is not None and comparison_time > 0:
                 ratio = current_response_time / comparison_time
                 if ratio  1.5: # 50% mais lento
                     response_time_shift = 'slower'

Comparar complexidade linguística (se aplicável)
        complexity_shift = 'stable'
        current_complexity = current_message.get('analysis', {}).get('linguistic_analysis', {}).get('style_metrics', {}).get('complexity')
        previous_complexity = previous_message.get('analysis', {}).get('linguistic_analysis', {}).get('style_metrics', {}).get('complexity')

        if current_complexity is not None and previous_complexity is not None:
             complexity_diff = current_complexity - previous_complexity
             if abs(complexity_diff) > 0.2: # Mudança significativa (> 0.2)
                 complexity_shift = 'more_complex' if complexity_diff > 0 else 'less_complex'

Determinar ajuste de tom com base nas mudanças
        suggested_tone_adjustment = 'maintain_neutrality' # Padrão

Lógica de ajuste (pode ser refinada com regras mais complexas)
        if emotion_shift == 'more_negative' or response_time_shift == 'slower':
            suggested_tone_adjustment = 'increase_empathy' # Resposta negativa ou mais lenta -> Aumentar empatia
        elif emotion_shift == 'more_positive' or response_time_shift == 'faster':
             suggested_tone_adjustment = 'match_positivity' # Resposta positiva ou mais rápida -> Espelhar positividade
        elif complexity_shift == 'less_complex' and emotion_shift != 'more_positive':
             suggested_tone_adjustment = 'simplify_and_support' # Simplificação pode indicar dificuldade -> Simplificar e apoiar
        elif emotion_shift == 'more_aroused':
             suggested_tone_adjustment = 'calm_and_structured' # Mais agitado -> Oferecer calma e estrutura

Considerar o perfil (e.g., se a pessoa tem baixa abertura emocional, evitar "match_positivity" muito intenso)
        emotional_openness = profile.get('psychological_profile', {}).get('emotional_tendencies', {}).get('emotional_openness', {}).get('level', 'média')
        if emotional_openness == 'baixa' and suggested_tone_adjustment == 'match_positivity':
            suggested_tone_adjustment = 'maintain_neutrality' # Evitar excesso de positividade

        return {
            'suggested_tone_adjustment': suggested_tone_adjustment,
            'detected_shifts': {
                'emotion': emotion_shift,
                'response_time': response_time_shift,
                'complexity': complexity_shift
            },
            'reason': 'variation_detected'
        }


13. Estímulo de Validação Intermitente

Este módulo implementa um cronograma variável para oferecer validação, promovendo um reforço mais natural e menos previsível.

    def generate_intermittent_validation(self, profile_id, current_interaction_context):
        """
        Gera validação ou reforço positivo de forma intermitente,
        baseado em um cronograma variável e na qualidade da interação.
        """
        if not profile_id or profile_id not in self.profiles:
            return None

        profile = self.profiles[profile_id]

Obter dados de calibração de validação do perfil
        validation_calibration = profile.get('communication_patterns', {}).get('validation_calibration', {})
        last_validation_timestamp = validation_calibration.get('last_timestamp')
        target_interval_hours = validation_calibration.get('target_interval_hours', 24) # Padrão: 1 vez por dia
        base_probability = validation_calibration.get('base_probability', 0.3) # Padrão: 30% de chance base

Verificar se tempo suficiente passou desde a última validação
        time_passed = False
        if last_validation_timestamp:
            from datetime import datetime, timedelta
            last_time = datetime.fromisoformat(last_validation_timestamp)
            if datetime.now() > last_time + timedelta(hours=target_interval_hours * 0.8): # 80% do intervalo
                 time_passed = True
        else:
            time_passed = True # Primeira validação

        if not time_passed:
            return None # Ainda não é hora

Calcular probabilidade atual de validação
        current_probability = base_probability

Ajustar probabilidade com base no contexto atual (qualidade da última interação)
        last_message = current_interaction_context.get('last_message')
        if last_message:
Verificar se a última mensagem do alvo foi positiva
             behavior = last_message.get('analysis', {}).get('behavioral_indicators', {}).get('dominant_pattern', {}).get('pattern')
             attitude = last_message.get('analysis', {}).get('attitude', {}).get('primary_attitude')

             if behavior in ['proactive', 'collaborative'] or attitude == 'colaborativa':
                 current_probability *= 1.5 # Aumentar chance se última interação foi positiva
             elif behavior in ['avoidance', 'reactive'] or attitude in ['resistente', 'evasiva']:
                  current_probability *= 0.7 # Diminuir chance se última interação foi negativa

Limitar probabilidade
        current_probability = max(0.1, min(0.9, current_probability))

Decidir se validação será oferecida
        import random
        if random.random() > current_probability:
             return None # Não oferecer validação desta vez

Gerar frase de validação (variada)
        validation_templates = [
             "Aprecio muito sua perspectiva sobre este assunto.",
             "Sua contribuição tem sido realmente valiosa.",
             "Gosto da forma como você está abordando esta questão.",
             "É ótimo trabalhar com você neste projeto.",
             "Sua clareza na comunicação facilita muito nosso trabalho.",
             "Reconheço o esforço que você tem dedicado a isso.",
             "Sua opinião é sempre muito bem-vinda e considerada."
        ]

Selecionar validação específica se houver mérito detectado na última mensagem
        merit_recognition = self.generate_merit_based_recognition(last_message, profile_id) if last_message else None
        if merit_recognition and merit_recognition.get('recognition_phrases'):
             validation_phrase = random.choice(merit_recognition['recognition_phrases'])
        else:
Usar template genérico
             validation_phrase = random.choice(validation_templates)

Atualizar dados de calibração no perfil
        if 'communication_patterns' not in profile:
             profile['communication_patterns'] = {}

        profile'communication_patterns' = {
             'last_timestamp': datetime.now().isoformat(),
             'target_interval_hours': target_interval_hours, # Poderia ser ajustado dinamicamente
             'base_probability': base_probability # Poderia ser ajustado dinamicamente
        }
        self.update_profile(profile_id, profile)


        return {
            'validation_phrase': validation_phrase,
            'reason': 'intermittent_schedule_met'
        }
