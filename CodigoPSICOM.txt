    def create_new_profile(self, name, description="", additional_data=None):
        """Cria um novo perfil de pessoa para análise"""
        # Gerar ID único baseado no nome
        profile_id = re.sub(r'[^a-zA-Z0-9]', '_', name.lower())
        
        # Verificar se já existe perfil com esse ID
        if profile_id in self.profiles:
            # Gerar ID único adicionando um número
            base_id = profile_id
            counter = 1
            while profile_id in self.profiles:
                profile_id = f"{base_id}_{counter}"
                counter += 1
        
        # Estrutura base do perfil
        profile_data = {
            "profile_id": profile_id,
            "name": name,
            "description": description,
            "created_at": datetime.now().isoformat(),
            "last_updated": datetime.now().isoformat(),
            "analyses": [],
            "behavioral_patterns": {
                "communication_style": {},
                "emotional_patterns": {},
                "response_patterns": {},
                "linguistic_patterns": {},
                "topic_preferences": {},
                "persuasion_indicators": {}
            },
            "communication_stats": {},
            "psychological_profile": {
                "personality_traits": {},
                "cognitive_patterns": {},
                "emotional_tendencies": {},
                "motivation_factors": {},
                "communication_preferences": {}
            },
            "recommendations": {
                "general": [],
                "by_topic": {},
                "by_context": {}
            }
        }
        
        # Adicionar dados adicionais se fornecidos
        if additional_data and isinstance(additional_data, dict):
            for key, value in additional_data.items():
                if key not in profile_data:
                    profile_data[key] = value
        
        # Salvar perfil
        profile_path = os.path.join(self.config["PROFILES_DIR"], f"{profile_id}.json")
        
        try:
            with open(profile_path, 'w', encoding='utf-8') as f:
                json.dump(profile_data, f, ensure_ascii=False, indent=2)
            
            # Adicionar à coleção de perfis
            self.profiles[profile_id] = profile_data
            
            logger.info(f"Novo perfil criado: {profile_id} ({name})")
            return profile_id
        except Exception as e:
            logger.error(f"Erro ao criar perfil {name}: {e}")
            return None
    
    def update_profile(self, profile_id, updated_data):
        """Atualiza um perfil existente"""
        if profile_id not in self.profiles:
            logger.error(f"Perfil não encontrado: {profile_id}")
            return False
        
        # Fazer backup do perfil atual
        backup_dir = os.path.join(self.config["PROFILES_DIR"], "backups")
        os.makedirs(backup_dir, exist_ok=True)
        
        backup_file = f"{profile_id}_{datetime.now().strftime('%Y%m%d%H%M%S')}.json"
        backup_path = os.path.join(backup_dir, backup_file)
        
        try:
            with open(backup_path, 'w', encoding='utf-8') as f:
                json.dump(self.profiles[profile_id], f, ensure_ascii=False, indent=2)
            
            # Atualizar perfil com novos dados
            for key, value in updated_data.items():
                # Não substituir campos estruturais
                if key not in ["profile_id", "created_at", "analyses"]:
                    self.profiles[profile_id][key] = value
            
            # Atualizar timestamp
            self.profiles[profile_id]["last_updated"] = datetime.now().isoformat()
            
            # Salvar perfil atualizado
            profile_path = os.path.join(self.config["PROFILES_DIR"], f"{profile_id}.json")
            
            with open(profile_path, 'w', encoding='utf-8') as f:
                json.dump(self.profiles[profile_id], f, ensure_ascii=False, indent=2)
            
            logger.info(f"Perfil atualizado: {profile_id}")
            return True
        
        except Exception as e:
            logger.error(f"Erro ao atualizar perfil {profile_id}: {e}")
            return False
    
    def delete_profile(self, profile_id):
        """Exclui um perfil existente"""
        if profile_id not in self.profiles:
            logger.error(f"Perfil não encontrado: {profile_id}")
            return False
        
        # Fazer backup do perfil
        backup_dir = os.path.join(self.config["PROFILES_DIR"], "deleted")
        os.makedirs(backup_dir, exist_ok=True)
        
        backup_file = f"{profile_id}_{datetime.now().strftime('%Y%m%d%H%M%S')}.json"
        backup_path = os.path.join(backup_dir, backup_file)
        
        try:
            with open(backup_path, 'w', encoding='utf-8') as f:
                json.dump(self.profiles[profile_id], f, ensure_ascii=False, indent=2)
            
            # Remover arquivo do perfil
            profile_path = os.path.join(self.config["PROFILES_DIR"], f"{profile_id}.json")
            
            if os.path.exists(profile_path):
                os.remove(profile_path)
            
            # Remover da coleção de perfis
            del self.profiles[profile_id]
            
            logger.info(f"Perfil excluído: {profile_id}")
            
            # Se for o perfil atual, limpar referência
            if self.current_profile == profile_id:
                self.current_profile = None
            
            return True
        
        except Exception as e:
            logger.error(f"Erro ao excluir perfil {profile_id}: {e}")
            return False

    def load_whatsapp_chat(self, file_path):
        """Carrega e processa o arquivo de chat do WhatsApp exportado"""
        try:
            with open(file_path, 'r', encoding='utf-8') as file:
                chat_text = file.read()
            
            # Padrão para mensagens do WhatsApp
            message_pattern = re.compile(r'\[(\d{2}/\d{2}/\d{4}), (\d{2}:\d{2}:\d{2})\] ([^:]+): (.*)', re.DOTALL)
            
            messages = []
            
            # Iterar pelos matches encontrados
            for match in message_pattern.finditer(chat_text):
                date_str, time_str, author, content = match.groups()
                
                # Converter strings de data e hora para objeto datetime
                date_time_str = f"{date_str} {time_str}"
                dt = datetime.strptime(date_time_str, '%d/%m/%Y %H:%M:%S')
                
                # Detectar tipo de conteúdo (texto, áudio, imagem, etc.)
                content_type, media_file = self.detect_content_type(content)
                
                # Preparar estrutura de dados para a mensagem
                message = {
                    'id': f"msg_{len(messages)}_{dt.strftime('%Y%m%d%H%M%S')}",
                    'datetime': dt.isoformat(),
                    'date': dt.date().isoformat(),
                    'time': dt.time().isoformat(),
                    'timestamp': dt.timestamp(),
                    'author': author.strip(),
                    'content': content.strip(),
                    'content_type': content_type,
                    'media_file': media_file,
                    'is_target': self.is_target_person(author.strip()),
                    'media_processed': False,
                    'transcription': None,
                    'analysis': {
                        'emotion': None,
                        'attitude': None,
                        'themes': [],
                        'behavioral_indicators': {},
                        'psychological_indicators': {},
                        'linguistic_analysis': {}
                    },
                    'context': {
                        'previous_messages': [],
                        'conversation_state': None,
                        'user_observations': None
                    }
                }
                
                messages.append(message)
            
            logger.info(f"Conversa carregada: {len(messages)} mensagens")
            return messages
        
        except Exception as e:
            logger.error(f"Erro ao carregar conversa: {e}")
            return []
    
    def detect_content_type(self, content):
        """Detecta o tipo de conteúdo e arquivo de mídia associado"""
        content = content.strip()
        
        # Padrões para diferentes tipos de mídia
        audio_pattern = re.compile(r'(PTT-\d{8}-WA\d{4}\.opus)')
        image_pattern = re.compile(r'(IMG-\d{8}-WA\d{4}\.(jpg|jpeg|png))')
        video_pattern = re.compile(r'(VID-\d{8}-WA\d{4}\.(mp4|3gp))')
        document_pattern = re.compile(r'([^<>:"/\\|?*]+\.(pdf|doc|docx|xls|xlsx|txt))')
        
        # Verificar áudio
        audio_match = audio_pattern.search(content)
        if audio_match:
            return 'audio', audio_match.group(1)
        
        # Verificar imagem
        image_match = image_pattern.search(content)
        if image_match:
            return 'image', image_match.group(1)
        
        # Verificar vídeo
        video_match = video_pattern.search(content)
        if video_match:
            return 'video', video_match.group(1)
        
        # Verificar documento
        document_match = document_pattern.search(content)
        if document_match:
            return 'document', document_match.group(1)
        
        # Texto comum
        return 'text', None
    
    def is_target_person(self, author):
        """Verifica se o autor da mensagem é a pessoa-alvo"""
        if not self.current_profile or not self.profiles.get(self.current_profile):
            return False
        
        target_name = self.profiles[self.current_profile]["name"]
        return target_name.lower() in author.lower()
    
    def process_media_files(self, conversation, source_dir, progress_callback=None):
        """Processa todos os arquivos de mídia na conversa"""
        # Contar total de arquivos de mídia
        media_files = [msg for msg in conversation if msg['content_type'] != 'text' and msg['media_file']]
        total_media = len(media_files)
        processed = 0
        
        if total_media == 0:
            logger.info("Nenhum arquivo de mídia para processar")
            return conversation
        
        logger.info(f"Processando {total_media} arquivos de mídia")
        
        # Criar diretório para mídia do perfil atual
        profile_media_dir = os.path.join(self.config["DATA_DIR"], self.current_profile, "media")
        os.makedirs(profile_media_dir, exist_ok=True)
        
        # Processar cada arquivo de mídia
        for msg in media_files:
            try:
                media_file = msg['media_file']
                source_path = os.path.join(source_dir, media_file)
                
                # Verificar se o arquivo existe
                if not os.path.exists(source_path):
                    logger.warning(f"Arquivo não encontrado: {source_path}")
                    continue
                
                # Definir caminho de destino
                dest_path = os.path.join(profile_media_dir, media_file)
                
                # Copiar arquivo se não existir
                if not os.path.exists(dest_path):
                    shutil.copy2(source_path, dest_path)
                    logger.debug(f"Arquivo copiado: {media_file}")
                
                # Processar conforme o tipo de mídia
                if msg['content_type'] == 'audio':
                    self.process_audio_file(msg, dest_path)
                elif msg['content_type'] == 'image':
                    self.process_image_file(msg, dest_path)
                elif msg['content_type'] == 'video':
                    self.process_video_file(msg, dest_path)
                
                # Marcar como processado
                msg['media_processed'] = True
                
                # Atualizar progresso
                processed += 1
                if progress_callback:
                    progress_callback(processed / total_media)
                    
            except Exception as e:
                logger.error(f"Erro ao processar mídia {msg.get('media_file', 'desconhecida')}: {e}")
        
        logger.info(f"Processamento de mídia concluído: {processed}/{total_media} arquivos")
        return conversation
    
    def process_audio_file(self, message, audio_path):
        """Processa um arquivo de áudio: converte, transcreve e analisa"""
        try:
            # Converter de .opus para .wav
            wav_path = audio_path.replace('.opus', '.wav')
            
            if not os.path.exists(wav_path):
                self.convert_audio_to_wav(audio_path, wav_path)
            
            # Transcrever o áudio
            transcription = self.transcribe_audio(wav_path)
            message['transcription'] = transcription
            
            # Analisar emoções no áudio
            emotion_analysis = self.analyze_audio_emotion(wav_path)
            message['analysis']['emotion'] = emotion_analysis
            
            logger.debug(f"Áudio processado: {os.path.basename(audio_path)}")
            return True
            
        except Exception as e:
            logger.error(f"Erro ao processar áudio {os.path.basename(audio_path)}: {e}")
            return False
    
    def convert_audio_to_wav(self, input_path, output_path):
        """Converte um arquivo de áudio para o formato WAV usando FFmpeg"""
        try:
            # Construir comando
            ffmpeg_cmd = [
                self.config["FFMPEG_PATH"],
                "-i", input_path,
                "-ar", "16000",
                "-ac", "1",
                "-y",
                output_path
            ]
            
            # Executar comando
            result = subprocess.run(
                ffmpeg_cmd,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                check=True
            )
            
            logger.debug(f"Áudio convertido: {os.path.basename(input_path)} -> {os.path.basename(output_path)}")
            return True
            
        except subprocess.CalledProcessError as e:
            logger.error(f"Erro ao converter áudio com FFmpeg: {e}")
            logger.debug(f"Saída de erro do FFmpeg: {e.stderr.decode('utf-8', errors='ignore')}")
            return False
        
        except Exception as e:
            logger.error(f"Erro ao converter áudio: {e}")
            return False
    
    def transcribe_audio(self, audio_path):
        """Transcreve o áudio usando Whisper"""
        try:
            # Importação condicional para permitir execução sem whisper instalado
            import whisper
            
            logger.info(f"Transcrevendo áudio: {os.path.basename(audio_path)}")
            
            # Carregar modelo
            model = whisper.load_model(self.config["WHISPER_MODEL"])
            
            # Transcrever
            result = model.transcribe(
                audio_path, 
                language=self.config["DEFAULT_LANGUAGE"],
                fp16=False
            )
            
            logger.debug(f"Transcrição concluída: {os.path.basename(audio_path)}")
            return result["text"]
            
        except ImportError:
            logger.warning("Biblioteca Whisper não encontrada. Usando modo simulado.")
            return f"[Simulação de transcrição para {os.path.basename(audio_path)}]"
            
        except Exception as e:
            logger.error(f"Erro ao transcrever áudio: {e}")
            return f"[Erro na transcrição: {str(e)}]"
    
    def analyze_audio_emotion(self, audio_path):
        """Analisa emoções no áudio usando OpenSMILE ou modelo personalizado"""
        # Verificar se OpenSMILE está configurado
        if self.config["OPENSMILE_PATH"] and os.path.exists(self.config["OPENSMILE_PATH"]):
            return self.analyze_with_opensmile(audio_path)
        else:
            # Usar método alternativo para análise
            return self.analyze_audio_features(audio_path)
    
    def analyze_with_opensmile(self, audio_path):
        """Analisa áudio usando OpenSMILE para extração de características emocionais"""
        try:
            # Arquivo temporário para saída
            output_file = os.path.join(
                self.config["TEMP_DIR"], 
                f"{os.path.basename(audio_path)}_egemaps.csv"
            )
            
            # Construir comando do OpenSMILE
            opensmile_cmd = [
                self.config["OPENSMILE_PATH"],
                "-C", self.config["OPENSMILE_CONFIG"],
                "-I", audio_path,
                "-O", output_file
            ]
            
            # Executar comando
            subprocess.run(
                opensmile_cmd,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                check=True
            )
            
            # Processar resultados
            emotion_data = self.process_opensmile_output(output_file)
            
            # Limpar arquivo temporário
            if os.path.exists(output_file):
                os.remove(output_file)
            
            return emotion_data
            
        except subprocess.CalledProcessError as e:
            logger.error(f"Erro ao executar OpenSMILE: {e}")
            return self.simulate_emotion_analysis(audio_path)
            
        except Exception as e:
            logger.error(f"Erro na análise de emoção com OpenSMILE: {e}")
            return self.simulate_emotion_analysis(audio_path)
    
    def process_opensmile_output(self, csv_path):
        """Processa a saída do OpenSMILE para extrair características emocionais"""
        try:
            # Ler CSV
            df = pd.read_csv(csv_path, sep=';')
            
            # Principais características emocionais
            features = {
                # Energia/intensidade
                'intensity': float(df['loudness_sma3_amean'].values[0]),
                
                # Pitch médio e variação (relacionados à tensão e expressividade)
                'pitch_mean': float(df['F0semitoneFrom27.5Hz_sma3nz_amean'].values[0]),

    def map_features_to_emotions(self, features):
        """Mapeia características acústicas para estados emocionais"""
        # Calcular intensidade emocional geral (0-10)
        intensity = min(10, max(1, features['intensity'] * 10))
        
        # Calcular eixos emocionais básicos
        # Valência (negativa-positiva)
        if features['pitch_mean'] > 100 and features['jitter'] < 0.01:
            valence = 0.7  # Provavelmente positiva
        elif features['jitter'] > 0.02 and features['intensity'] < 0.5:
            valence = 0.3  # Provavelmente negativa
        else:
            valence = 0.5  # Neutra
        
        # Ativação (calma-excitada)
        if features['intensity'] > 0.7 or features['speech_rate'] > 0.8:
            arousal = 0.8  # Alta ativação
        elif features['intensity'] < 0.4 and features['speech_rate'] < 0.5:
            arousal = 0.3  # Baixa ativação
        else:
            arousal = 0.5  # Média
        
        # Tensão (relaxada-tensa)
        if features['jitter'] > 0.02 or features['pitch_range'] < 0.2:
            tension = 0.7  # Tensa
        else:
            tension = 0.4  # Relaxada
        
        # Identificar emoções primárias e secundárias
        emotions = []
        
        # Alta valência, alta ativação
        if valence > 0.6 and arousal > 0.6:
            emotions.append("entusiasmo")
        # Alta valência, baixa ativação
        elif valence > 0.6 and arousal < 0.4:
            emotions.append("contentamento")
        # Baixa valência, alta ativação
        elif valence < 0.4 and arousal > 0.6:
            emotions.append("irritação" if tension > 0.6 else "ansiedade")
        # Baixa valência, baixa ativação
        elif valence < 0.4 and arousal < 0.4:
            emotions.append("tristeza")
        # Baixa valência, média ativação, alta tensão
        elif valence < 0.45 and arousal > 0.4 and arousal < 0.6 and tension > 0.6:
            emotions.append("frustração")
        # Média valência, alta tensão
        elif valence > 0.4 and valence < 0.6 and tension > 0.6:
            emotions.append("tensão")
        # Média valência, média ativação, baixa tensão
        elif valence > 0.4 and valence < 0.6 and arousal > 0.4 and arousal < 0.6 and tension < 0.5:
            emotions.append("neutro")
        
        # Detectar hesitação
        if features['pitch_range'] > 0.6 and features['speech_rate'] < 0.6:
            emotions.append("hesitação")
        
        # Detectar confiança
        if features['intensity'] > 0.6 and features['jitter'] < 0.01 and tension < 0.5:
            emotions.append("confiança")
        
        # Se nenhuma emoção foi identificada
        if not emotions:
            emotions.append("neutro")
        
        # Calcular nível de autenticidade emocional (0-100)
        authenticity = 85 - (features['jitter'] * 1000) - (abs(0.5 - tension) * 50)
        authenticity = max(0, min(100, authenticity))
        
        # Resultado final
        return {
            'primary_emotion': emotions[0],
            'emotional_blend': " + ".join(emotions),
            'intensity': round(intensity, 1),
            'valence': round(valence * 10, 1),    # Escala 0-10 para facilitar interpretação
            'arousal': round(arousal * 10, 1),    # Escala 0-10
            'tension': round(tension * 10, 1),    # Escala 0-10
            'authenticity': round(authenticity),  # Porcentagem
            'emotions': emotions
        }
    
    def analyze_audio_features(self, audio_path):
        """Analisa características do áudio usando bibliotecas alternativas"""
        try:
            # Importação condicional
            import librosa
            import numpy as np
            
            # Carregar áudio
            y, sr = librosa.load(audio_path, sr=None)
            
            # Extrair características
            # Energia/intensidade
            rms = librosa.feature.rms(y=y)[0]
            intensity = np.mean(rms) / 0.1  # Normalizar
            
            # Pitch estimado e variação
            pitches, magnitudes = librosa.piptrack(y=y, sr=sr)
            pitch_mean = np.mean(pitches[magnitudes > np.median(magnitudes)])
            pitch_range = np.std(pitches[magnitudes > np.median(magnitudes)])
            
            # Taxa de cruzamento zero (aspereza)
            zcr = librosa.feature.zero_crossing_rate(y)[0]
            jitter = np.std(zcr) / np.mean(zcr) if np.mean(zcr) > 0 else 0
            
            # Medidas espectrais
            spectral_contrast = np.mean(librosa.feature.spectral_contrast(y=y, sr=sr))
            
            # Convertendo para o mesmo formato usado pelo OpenSMILE
            features = {
                'intensity': min(1.0, max(0.1, intensity)),
                'pitch_mean': pitch_mean if not np.isnan(pitch_mean) else 100,
                'pitch_range': min(1.0, max(0.1, pitch_range / 100)),
                'jitter': min(0.05, max(0.001, jitter)),
                'shimmer': min(0.1, max(0.01, np.std(rms) / (np.mean(rms) + 1e-10))),
                'speech_rate': min(1.0, max(0.1, np.mean(zcr))),
                'formant1_mean': 500,  # Valor aproximado
                'formant1_bandwidth': 100  # Valor aproximado
            }
            
            # Mapear para emoções
            emotion_analysis = self.map_features_to_emotions(features)
            emotion_analysis['raw_features'] = features
            
            return emotion_analysis
            
        except ImportError:
            logger.warning("Bibliotecas para análise de áudio não encontradas. Usando simulação.")
            return self.simulate_emotion_analysis(audio_path)
            
        except Exception as e:
            logger.error(f"Erro na análise alternativa de áudio: {e}")
            return self.simulate_emotion_analysis(audio_path)

    def analyze_conversation(self, conversation, progress_callback=None):
        """Realiza análise completa da conversa"""
        total_messages = len(conversation)
        processed = 0
        
        # Identificar mensagens do perfil alvo
        target_messages = [msg for msg in conversation if msg['is_target']]
        logger.info(f"Analisando {len(target_messages)} mensagens do perfil alvo de {total_messages} totais")
        
        # Análise de contexto para toda a conversa
        conversation_context = self.analyze_conversation_context(conversation)
        
        # Processar cada mensagem do alvo
        for i, msg in enumerate(conversation):
            # Pular mensagens que não são do alvo, exceto para contexto
            if not msg['is_target']:
                # Adicionar contexto mínimo para mensagens que não são do alvo
                self.add_context_to_message(msg, conversation, i, minimal=True)
                continue
            
            # Obter texto para análise
            text = msg['transcription'] if msg['content_type'] == 'audio' and msg['transcription'] else msg['content']
            
            # Adicionar contexto da conversa
            self.add_context_to_message(msg, conversation, i)
            
            # Análise linguística
            msg['analysis']['linguistic_analysis'] = self.analyze_text_linguistics(text)
            
            # Análise de atitude
            msg['analysis']['attitude'] = self.classify_attitude(text, msg)
            
            # Identificação de temas
            msg['analysis']['themes'] = self.identify_themes(text)
            
            # Análise comportamental
            msg['analysis']['behavioral_indicators'] = self.analyze_behavioral_patterns(text, msg)
            
            # Análise psicológica
            msg['analysis']['psychological_indicators'] = self.analyze_psychological_patterns(text, msg)
            
            # Atualizar progresso
            processed += 1
            if progress_callback:
                progress_callback(processed / len(target_messages))
        
        # Análise cruzada e identificação de padrões gerais
        analysis_summary = self.generate_analysis_summary(conversation, conversation_context)
        
        logger.info("Análise de conversa concluída")
        return conversation, analysis_summary
    
    def add_context_to_message(self, message, conversation, message_index, minimal=False):
        """Adiciona informações de contexto a uma mensagem"""
        # Determinar janela de contexto
        context_size = self.config["CONTEXT_WINDOW_SIZE"]
        start_idx = max(0, message_index - context_size)
        
        # Coletar mensagens anteriores para contexto
        previous_messages = []
        for i in range(start_idx, message_index):
            prev_msg = conversation[i]
            prev_text = prev_msg['transcription'] if prev_msg['content_type'] == 'audio' and prev_msg['transcription'] else prev_msg['content']
            
            previous_messages.append({
                'id': prev_msg['id'],
                'author': prev_msg['author'],
                'content': prev_text,
                'timestamp': prev_msg['timestamp'],
                'is_target': prev_msg['is_target']
            })
        
        message['context']['previous_messages'] = previous_messages
        
        # Para mensagens que não são do alvo, não precisamos de contexto adicional
        if minimal:
            return
        
        # Determinar estado da conversa
        if previous_messages:
            # Verificar se a mensagem atual é uma resposta
            is_response = not previous_messages[-1]['is_target']
            
            # Calcular tempo de resposta se for uma resposta
            response_time = None
            if is_response:
                prev_timestamp = previous_messages[-1]['timestamp']
                current_timestamp = message['timestamp']
                response_time = (current_timestamp - prev_timestamp) / 60  # minutos
            
            # Determinar estado (iniciando conversa, respondendo, continuando)
            if not previous_messages:
                conversation_state = "iniciando"
            elif is_response:
                conversation_state = "respondendo"
            else:
                conversation_state = "continuando"
            
            message['context']['conversation_state'] = {
                'state': conversation_state,
                'is_response': is_response,
                'response_time': response_time,
                'previous_topic': self.identify_main_topic(previous_messages)
            }
    
    def identify_main_topic(self, messages):
        """Identifica o tópico principal de uma sequência de mensagens"""
        if not messages:
            return None
        
        # Coletar todos os temas mencionados
        all_themes = []
        for msg in messages:
            if isinstance(msg, dict) and 'content' in msg:
                themes = self.identify_themes(msg['content'])
                all_themes.extend(themes)
        
        # Se não houver temas identificados
        if not all_themes:
            return None
        
        # Contar ocorrências
        theme_counter = Counter(all_themes)
        
        # Retornar o tema mais comum
        return theme_counter.most_common(1)[0][0]
    
    def analyze_conversation_context(self, conversation):
        """Analisa o contexto geral da conversa"""
        # Extrair informações básicas
        total_messages = len(conversation)
        target_messages = [msg for msg in conversation if msg['is_target']]
        user_messages = [msg for msg in conversation if not msg['is_target']]
        
        # Calcular estatísticas básicas
        message_ratio = len(target_messages) / total_messages if total_messages > 0 else 0
        
        # Calcular média de palavras por mensagem
        target_word_counts = []
        for msg in target_messages:
            text = msg['transcription'] if msg['content_type'] == 'audio' and msg['transcription'] else msg['content']
            word_count = len(text.split())
            target_word_counts.append(word_count)
        
        avg_words_per_message = sum(target_word_counts) / len(target_word_counts) if target_word_counts else 0
        
        # Calcular tempos de resposta
        response_times = []
        for i in range(1, len(conversation)):
            curr_msg = conversation[i]
            prev_msg = conversation[i-1]
            
            # Se a mensagem atual é do alvo e a anterior não é
            if curr_msg['is_target'] and not prev_msg['is_target']:
                time_diff = datetime.fromisoformat(curr_msg['datetime']) - datetime.fromisoformat(prev_msg['datetime'])
                response_times.append(time_diff.total_seconds() / 60)  # minutos
        
        avg_response_time = sum(response_times) / len(response_times) if response_times else None
        
        # Determinar janela temporal da conversa
        if conversation:
            start_time = datetime.fromisoformat(conversation[0]['datetime'])
            end_time = datetime.fromisoformat(conversation[-1]['datetime'])
            duration = (end_time - start_time).total_seconds() / 3600  # horas
        else:
            start_time = None
            end_time = None
            duration = None
        
        # Identificar distribuição de tipos de conteúdo
        content_types = Counter([msg['content_type'] for msg in target_messages])
        
        # Construir objeto de contexto
        context = {
            'total_messages': total_messages,
            'target_messages': len(target_messages),
            'user_messages': len(user_messages),
            'message_ratio': message_ratio,
            'avg_words_per_message': avg_words_per_message,
            'avg_response_time': avg_response_time,
            'content_type_distribution': dict(content_types),
            'temporal_window': {
                'start_time': start_time.isoformat() if start_time else None,
                'end_time': end_time.isoformat() if end_time else None,
                'duration_hours': duration
            }
        }
        
        return context
    
    def analyze_text_linguistics(self, text):
        """Realiza análise linguística do texto"""
        # Análise básica
        word_count = len(text.split())
        sentence_count = len(re.split(r'[.!?]+', text))
        avg_word_length = sum(len(word) for word in text.split()) / word_count if word_count > 0 else 0
        
        # Detecção de formalidade
        formal_indicators = ['conforme', 'solicito', 'informo', 'cordialmente', 'atenciosamente', 
                           'prezado', 'senhor', 'senhora', 'doutor', 'doutora']
        informal_indicators = ['oi', 'olá', 'beleza', 'valeu', 'blz', 'tranquilo', 'cara', 'mano', 'véi']
        
        formal_count = sum(1 for word in formal_indicators if word in text.lower())
        informal_count = sum(1 for word in informal_indicators if word in text.lower())
        
        formality_score = (formal_count - informal_count) / (formal_count + informal_count + 1)
        formality_score = max(-1, min(1, formality_score))  # Normalizar entre -1 e 1
        
        # Detecção de complexidade
        complex_structures = ['entretanto', 'portanto', 'consequentemente', 'ademais', 'conquanto',
                             'à medida que', 'tendo em vista', 'em virtude de', 'não obstante']
        complex_count = sum(1 for phrase in complex_structures if phrase in text.lower())
        
        complexity_score = min(1.0, (avg_word_length / 8) + (complex_count / 5) + (sentence_count / 10))
        
        # Detecção de tom
        positive_words = ['bom', 'ótimo', 'excelente', 'feliz', 'satisfeito', 'grato', 'obrigado',
                         'agradeço', 'perfeito', 'maravilhoso', 'sim', 'claro', 'concordo']
        negative_words = ['ruim', 'péssimo', 'infelizmente', 'triste', 'insatisfeito', 'problema',
                         'erro', 'falha', 'desculpe', 'não', 'nunca', 'jamais', 'discordo']
        neutral_words = ['ok', 'bem', 'talvez', 'possivelmente', 'provavelmente', 'entendo']
        
        positive_count = sum(1 for word in positive_words if word in text.lower())
        negative_count = sum(1 for word in negative_words if word in text.lower())
        neutral_count = sum(1 for word in neutral_words if word in text.lower())
        
        # Determinar tom predominante
        if positive_count > negative_count and positive_count > neutral_count:
            tone = "positivo"
        elif negative_count > positive_count and negative_count > neutral_count:
            tone = "negativo"
        else:
            tone = "neutro"
        
        # Detectar hedging (linguagem evasiva/hesitante)
        hedging_phrases = ['eu acho', 'talvez', 'possivelmente', 'provavelmente', 'aparentemente',
                          'não tenho certeza', 'não sei', 'pode ser', 'quem sabe', 'vou ver']
        hedging_count = sum(1 for phrase in hedging_phrases if phrase in text.lower())
        hedging_score = min(1.0, hedging_count / 3)
        
        # Resultado da análise
        analysis = {
            'basic_metrics': {
                'word_count': word_count,
                'sentence_count': sentence_count,
                'avg_word_length': round(avg_word_length, 2)
            },
            'style_metrics': {
                'formality': round(formality_score, 2),
                'complexity': round(complexity_score, 2),
                'hedging': round(hedging_score, 2)
            },
            'tone': {
                'primary_tone': tone,
                'positive_score': round(positive_count / (word_count * 0.1) if word_count > 0 else 0, 2),
                'negative_score': round(negative_count / (word_count * 0.1) if word_count > 0 else 0, 2),
                'neutral_score': round(neutral_count / (word_count * 0.1) if word_count > 0 else 0, 2)
            }
        }
        
        return analysis
    
    def classify_attitude(self, text, message):

            ],
            'defensiva': [
                "não foi minha culpa", "eu não sabia", "ninguém me avisou",
                "não me responsabilizo", "não era minha função", "eu avisei",
                "já tinha dito", "não é minha responsabilidade", "fizeram sem me consultar"
            ]
        }
        
        # Contar ocorrências de indicadores
        attitude_scores = {}
        for attitude, indicators in attitude_indicators.items():
            count = sum(1 for phrase in indicators if phrase in text)
            attitude_scores[attitude] = count
        
        # Incorporar informações de emoção do áudio (se disponível)
        if message['content_type'] == 'audio' and message['analysis']['emotion']:
            emotion = message['analysis']['emotion']
            primary_emotion = emotion.get('primary_emotion', '')
            
            # Ajustar pontuações com base na emoção
            if primary_emotion == 'hesitação':
                attitude_scores['evasiva'] += 1
            elif primary_emotion == 'tensão':
                attitude_scores['defensiva'] += 0.5
                attitude_scores['resistente'] += 0.5
            elif primary_emotion == 'confiança':
                attitude_scores['autoritária'] += 0.5
                attitude_scores['colaborativa'] += 0.5
            elif primary_emotion == 'ansiedade':
                attitude_scores['evasiva'] += 0.5
                attitude_scores['submissa'] += 0.5
        
        # Determinar a atitude predominante
        max_score = max(attitude_scores.values()) if attitude_scores else 0
        
        if max_score > 0:
            # Lista de atitudes com a pontuação máxima
            primary_attitudes = [attitude for attitude, score in attitude_scores.items() if score == max_score]
            primary_attitude = primary_attitudes[0]  # Pegar a primeira se houver empate
        else:
            # Analisar com base em características linguísticas
            linguistic = message['analysis'].get('linguistic_analysis', {})
            
            if linguistic:
                formality = linguistic.get('style_metrics', {}).get('formality', 0)
                hedging = linguistic.get('style_metrics', {}).get('hedging', 0)
                tone = linguistic.get('tone', {}).get('primary_tone', 'neutro')
                
                if hedging > 0.5:
                    primary_attitude = 'evasiva'
                elif formality > 0.5 and tone == 'positivo':
                    primary_attitude = 'colaborativa'
                elif formality < -0.3 and tone == 'negativo':
                    primary_attitude = 'resistente'
                elif formality > 0.7:
                    primary_attitude = 'formal'
                else:
                    primary_attitude = 'neutra'
            else:
                primary_attitude = 'neutra'
        
        # Calcular força da atitude (0-10)
        if max_score > 0:
            attitude_strength = min(10, max_score * 3)
        else:
            attitude_strength = 3  # Valor médio-baixo para atitudes inferidas indiretamente
        
        # Criar resultado detalhado
        result = {
            'primary_attitude': primary_attitude,
            'strength': round(attitude_strength, 1),
            'scores': {k: round(v, 2) for k, v in attitude_scores.items() if v > 0}
        }
        
        return result

    def analyze_behavioral_patterns(self, text, message):
        """Analisa padrões comportamentais na mensagem"""
        # Obter contexto e análises já realizadas
        attitude = message.get('analysis', {}).get('attitude', {})
        linguistic = message.get('analysis', {}).get('linguistic_analysis', {})
        emotion = message.get('analysis', {}).get('emotion', {})
        
        # Indicadores comportamentais a detectar
        behavioral_indicators = {}
        
        # 1. Estratégias de evitação
        evasion_indicators = [
            "vou ver", "preciso verificar", "não sei dizer", "vamos pensar", 
            "depois falamos", "mais tarde vejo", "não tenho essa informação"
        ]
        
        evasion_score = sum(1 for phrase in evasion_indicators if phrase in text.lower())
        
        if attitude.get('primary_attitude') == 'evasiva':
            evasion_score += 1
        
        if linguistic.get('style_metrics', {}).get('hedging', 0) > 0.5:
            evasion_score += 1
        
        if emotion.get('primary_emotion') in ['hesitação', 'ansiedade']:
            evasion_score += 1
        
        behavioral_indicators['avoidance_strategy'] = {
            'score': min(10, evasion_score * 2),
            'present': evasion_score > 1
        }
        
        # 2. Transferência de responsabilidade
        responsibility_shifting = [
            "não é minha responsabilidade", "não depende de mim", "eles que",
            "não fui eu", "culpa", "responsável é", "não me foi informado",
            "ninguém me avisou", "não recebi", "o pessoal do", "o pessoal da"
        ]
        
        shift_score = sum(1 for phrase in responsibility_shifting if phrase in text.lower())
        
        if attitude.get('primary_attitude') == 'defensiva':
            shift_score += 1
        
        behavioral_indicators['responsibility_shifting'] = {
            'score': min(10, shift_score * 2),
            'present': shift_score > 0
        }
        
        # 3. Busca de aprovação
        approval_seeking = [
            "o que você acha", "está bom assim", "concorda", "por favor",
            "se você preferir", "se achar melhor", "como você quiser", "confirme se"
        ]
        
        approval_score = sum(1 for phrase in approval_seeking if phrase in text.lower())
        
        if attitude.get('primary_attitude') == 'submissa':
            approval_score += 1
        
        behavioral_indicators['approval_seeking'] = {
            'score': min(10, approval_score * 2),
            'present': approval_score > 0
        }
        
        # 4. Tentativa de controle
        control_indicators = [
            "você deve", "tem que", "precisa fazer", "é necessário que você",
            "exijo", "quero que", "faça", "imediatamente", "obrigatório"
        ]
        
        control_score = sum(1 for phrase in control_indicators if phrase in text.lower())
        
        if attitude.get('primary_attitude') == 'autoritária':
            control_score += 1
        
        behavioral_indicators['control_attempt'] = {
            'score': min(10, control_score * 2),
            'present': control_score > 0
        }
        
        # 5. Conexão emocional
        emotional_connection = [
            "sinto", "estou", "preocupado", "feliz", "triste", "ansioso",
            "cansado", "estressado", "emocionado", "nervoso", "animado"
        ]
        
        connection_score = sum(1 for phrase in emotional_connection if phrase in text.lower())
        
        # Verificar se há self-disclosure (revelação pessoal)
        personal_phrases = ["eu estou", "minha família", "meu filho", "minha filha", 
                          "não dormi", "estou doente", "minha saúde", "minha casa"]
        
        self_disclosure = sum(1 for phrase in personal_phrases if phrase in text.lower())
        
        behavioral_indicators['emotional_connection'] = {
            'score': min(10, connection_score + self_disclosure),
            'present': connection_score > 0 or self_disclosure > 0,
            'self_disclosure': self_disclosure > 0
        }
        
        # 6. Conflito interno (contradições na mesma mensagem)
        contradictions = {
            "sim, mas": 0,
            "quero, mas não posso": 0,
            "posso, mas não agora": 0,
            "gostaria, mas": 0
        }
        
        for phrase in contradictions.keys():
            if phrase in text.lower():
                contradictions[phrase] = 1
        
        # Verificar afirmações contraditórias
        has_yes = "sim" in text.lower() or "claro" in text.lower() or "posso" in text.lower()
        has_no = "não" in text.lower() or "impossível" in text.lower() or "difícil" in text.lower()
        
        behavioral_indicators['internal_conflict'] = {
            'score': min(10, sum(contradictions.values()) * 3 + (3 if has_yes and has_no else 0)),
            'present': sum(contradictions.values()) > 0 or (has_yes and has_no),
            'contradictions': {k: v > 0 for k, v in contradictions.items() if v > 0}
        }
        
        # 7. Reatividade vs. Proatividade
        proactive_indicators = [
            "já fiz", "providenciei", "resolvi", "organizei", "planejei",
            "preparei", "antecipei", "previ", "sugiro que", "recomendo"
        ]
        
        reactive_indicators = [
            "vou ver", "depois respondo", "quando precisar", "me avise", 
            "me informe", "se precisar", "se quiser", "se for necessário"
        ]
        
        proactive_score = sum(1 for phrase in proactive_indicators if phrase in text.lower())
        reactive_score = sum(1 for phrase in reactive_indicators if phrase in text.lower())
        
        behavioral_indicators['proactivity'] = {
            'score': min(10, proactive_score * 2 - reactive_score),
            'present': proactive_score > reactive_score,
            'reactive': reactive_score > proactive_score
        }
        
        # 8. Indicadores de status e hierarquia
        status_indicators = [
            "eu sou", "minha posição", "como responsável", "como chefe",
            "como coordenador", "minha função", "autoridade", "minha decisão",
            "determinei", "estabeleci", "defini", "doutor", "doutora", "senhor", "senhora"
        ]
        
        status_score = sum(1 for phrase in status_indicators if phrase in text.lower())
        
        behavioral_indicators['status_indication'] = {
            'score': min(10, status_score * 2),
            'present': status_score > 0
        }
        
        # Determinar padrão comportamental predominante
        behavior_scores = {
            'avoidance': behavioral_indicators['avoidance_strategy']['score'],
            'approval_seeking': behavioral_indicators['approval_seeking']['score'],
            'control': behavioral_indicators['control_attempt']['score'],
            'emotional': behavioral_indicators['emotional_connection']['score'],
            'conflict': behavioral_indicators['internal_conflict']['score'],
            'proactive': max(0, behavioral_indicators['proactivity']['score']),
            'reactive': max(0, -behavioral_indicators['proactivity']['score']),
            'status_focused': behavioral_indicators['status_indication']['score'],
            'responsibility_shifting': behavioral_indicators['responsibility_shifting']['score']
        }
        
        # Determinar comportamento predominante
        max_behavior = max(behavior_scores.items(), key=lambda x: x[1]) if behavior_scores else ('neutral', 0)
        
        behavioral_indicators['dominant_pattern'] = {
            'pattern': max_behavior[0] if max_behavior[1] > 3 else 'neutral',
            'score': max_behavior[1],
            'secondary_patterns': [k for k, v in behavior_scores.items() 
                                 if v > 3 and k != max_behavior[0]]
        }
        
        return behavioral_indicators
    
    def analyze_psychological_patterns(self, text, message):
        """Analisa padrões psicológicos na mensagem"""
        # Obter resultados de outras análises
        behavioral = message.get('analysis', {}).get('behavioral_indicators', {})
        attitude = message.get('analysis', {}).get('attitude', {})
        linguistic = message.get('analysis', {}).get('linguistic_analysis', {})
        emotion = message.get('analysis', {}).get('emotion', {})
        
        # Indicadores psicológicos a identificar
        psychological_indicators = {}
        
        # 1. Necessidade de controle
        control_need = 0
        
        if behavioral.get('control_attempt', {}).get('present', False):
            control_need += behavioral['control_attempt']['score'] / 2
        
        if attitude.get('primary_attitude') == 'autoritária':
            control_need += 3
        
        if behavioral.get('responsibility_shifting', {}).get('present', False):
            control_need += 2
        
        psychological_indicators['control_need'] = {
            'score': min(10, control_need),
            'level': 'alta' if control_need > 7 else ('média' if control_need > 4 else 'baixa')
        }
        
        # 2. Necessidade de aprovação
        approval_need = 0
        
        if behavioral.get('approval_seeking', {}).get('present', False):
            approval_need += behavioral['approval_seeking']['score'] / 2
        
        if attitude.get('primary_attitude') == 'submissa':
            approval_need += 3
        
        # Verificar preocupação com a opinião dos outros
        opinion_phrases = ["o que achou", "ficou bom", "gostou", "está aprovado",
                          "sua opinião", "você concorda", "parece bom", "está adequado"]
        
        opinion_score = sum(1 for phrase in opinion_phrases if phrase in text.lower())
        approval_need += opinion_score
        
        psychological_indicators['approval_need'] = {
            'score': min(10, approval_need),
            'level': 'alta' if approval_need > 7 else ('média' if approval_need > 4 else 'baixa')
        }
        
        # 3. Tolerância a incertezas
        uncertainty_tolerance = 10  # Começa alta, reduz com indicadores negativos
        
        # Fatores que reduzem a tolerância
        if behavioral.get('avoidance_strategy', {}).get('present', False):
            uncertainty_tolerance -= behavioral['avoidance_strategy']['score'] / 2
        
        if emotion.get('primary_emotion') in ['hesitação', 'ansiedade', 'tensão']:
            uncertainty_tolerance -= 3
        
        # Verificar necessidade de clareza/certeza
        certainty_phrases = ["preciso saber exatamente", "me informe com certeza",
                           "quero detalhes", "informação precisa", "confirmação",
                           "garantia", "certeza", "definição clara", "especificar"]
        
        certainty_score = sum(1 for phrase in certainty_phrases if phrase in text.lower())
        uncertainty_tolerance -= certainty_score
        
        psychological_indicators['uncertainty_tolerance'] = {
            'score': max(0, min(10, uncertainty_tolerance)),
            'level': 'alta' if uncertainty_tolerance > 7 else ('média' if uncertainty_tolerance > 4 else 'baixa')
        }
        
        # 4. Abertura emocional
        emotional_openness = 0
        
        if behavioral.get('emotional_connection', {}).get('present', False):
            emotional_openness += behavioral['emotional_connection']['score'] / 2
        
        if behavioral.get('emotional_connection', {}).get('self_disclosure', False):
            emotional_openness += 3
        
        # Verificar expressões emocionais explícitas
        emotion_words = ["feliz", "triste", "preocupado", "irritado", "ansioso",
                        "alegre", "animado", "frustrado", "estressado", "entusiasmado",
                        "sinto", "sentindo", "emocionado", "afetado", "tocado"]
        
        emotion_score = sum(1 for word in emotion_words if word in text.lower())
        emotional_openness += emotion_score
        
        psychological_indicators['emotional_openness'] = {
            'score': min(10, emotional_openness),
            'level': 'alta' if emotional_openness > 7 else ('média' if emotional_openness > 3 else 'baixa')
        }
        
        # 5. Flexibilidade cognitiva
        cognitive_flexibility = 10  # Começa alta, reduz com indicadores negativos
        
        if behavioral.get('dominant_pattern', {}).get('pattern') == 'control':
            cognitive_flexibility -= 3
        
        # Verificar linguagem rígida/inflexível
        rigid_phrases = ["sempre", "nunca", "impossível", "só existe uma maneira",
                       "tem que ser assim", "não tem como", "é a única forma",
                       "sem exceção", "absolutamente", "inquestionável"]
        
        rigid_score = sum(1 for phrase in rigid_phrases if phrase in text.lower())
        cognitive_flexibility -= rigid_score * 2
        
        # Verificar abertura a alternativas
        alternative_phrases = ["podemos tentar", "outra opção", "alternativa",
                             "flexível", "adaptar", "ajustar", "possibilidade",
                             "talvez possamos", "considerar diferentes"]
        
        alternative_score = sum(1 for phrase in alternative_phrases if phrase in text.lower())
        cognitive_flexibility += alternative_score
        
        psychological_indicators['cognitive_flexibility'] = {
            'score': max(0, min(10, cognitive_flexibility)),
            'level': 'alta' if cognitive_flexibility > 7 else ('média' if cognitive_flexibility > 4 else 'baixa')
        }
        
        # 6. Autonomia vs. Dependência
        autonomy_score = 5  # Valor neutro
        
        # Indicadores de autonomia
        if behavioral.get('dominant_pattern', {}).get('pattern') in ['proactive', 'control']:
            autonomy_score += 2
        
        if attitude.get('primary_attitude') == 'autoritária':
            autonomy_score += 2
        
        # Indicadores de dependência
        if behavioral.get('dominant_pattern', {}).get('pattern') in ['approval_seeking', 'reactive']:
            autonomy_score -= 2
        
        if attitude.get('primary_attitude') in ['submissa', 'evasiva']:
            autonomy_score -= 2
        
        # Verificar indicadores linguísticos
        autonomy_phrases = ["eu decidi", "vou fazer", "tomei a iniciativa", 
                          "resolvi", "determinei", "por conta própria", "independentemente"]
        
        dependency_phrases = ["você decide", "como você quiser", "preciso de orientação",
                            "me diga o que fazer", "aguardo sua decisão", "depende de você",
                            "não sei se posso", "preciso de permissão"]
        
        autonomy_phrase_score = sum(1 for phrase in autonomy_phrases if phrase in text.lower())
        dependency_phrase_score = sum(1 for phrase in dependency_phrases if phrase in text.lower())
        
        autonomy_score += autonomy_phrase_score - dependency_phrase_score
        
        psychological_indicators['autonomy'] = {
            'score': max(0, min(10, autonomy_score)),
            'orientation': 'autônomo' if autonomy_score > 7 else ('dependente' if autonomy_score < 3 else 'equilibrado')
        }
        
        # 7. Orientação temporal (presente, passado, futuro)
        present_indicators = ["agora", "neste momento", "hoje", "atualmente", "já", "nesse instante"]
        past_indicators = ["ontem", "antes", "anteriormente", "já foi", "tinha", "fiz", "fizemos"]
        future_indicators = ["amanhã", "depois", "mais tarde", "próximo", "futuramente", "será", "irei", "faremos"]
        
        present_count = sum(1 for word in present_indicators if word in text.lower())
        past_count = sum(1 for word in past_indicators if word in text.lower())
        future_count = sum(1 for word in future_indicators if word in text.lower())
        
        # Determinar orientação temporal predominante
        temporal_counts = {'presente': present_count, 'passado': past_count, 'futuro': future_count}
        max_temporal = max(temporal_counts.items(), key=lambda x: x[1])
        
        psychological_indicators['temporal_orientation'] = {
            'primary': max_temporal[0] if max_temporal[1] > 0 else 'presente',
            'counts': temporal_counts
        }
        
        # 8. Nível de vulnerabilidade percebida
        vulnerability = 0
        
        if emotional_openness > 5:
            vulnerability += 2
        
        if behavioral.get('emotional_connection', {}).get('self_disclosure', False):
            vulnerability += 3
        
        if attitude.get('primary_attitude') in ['submissa', 'evasiva']:
            vulnerability += 2
        
        # Verificar exposição de problemas pessoais
        vulnerability_phrases = ["não consegui", "estou com dificuldade", "preciso de ajuda",
                               "não estou bem", "estou doente", "problema pessoal",
                               "estou sofrendo", "não sei como", "estou perdido"]
        
        vulnerability_score = sum(1 for phrase in vulnerability_phrases if phrase in text.lower())
        vulnerability += vulnerability_score * 2
        
        psychological_indicators['vulnerability'] = {
            'score': min(10, vulnerability),
            'level': 'alta' if vulnerability > 7 else ('média' if vulnerability > 3 else 'baixa')
        }
        
        # Identificar perfil psicológico predominante
        profile_indicators = {
            'controlador': psychological_indicators['control_need']['score'],
            'aprovação-dependente': psychological_indicators['approval_need']['score'],
            'incerto': 10 - psychological_indicators['uncertainty_tolerance']['score'],
            'emocionalmente_aberto': psychological_indicators['emotional_openness']['score'],
            'rígido': 10 - psychological_indicators['cognitive_flexibility']['score'],
            'autônomo': psychological_indicators['autonomy']['score'],
            'vulnerável': psychological_indicators['vulnerability']['score']
        }
        
        # Identificar características dominantes (acima de 7)
        dominant_traits = {k: v for k, v in profile_indicators.items() if v > 7}
        secondary_traits = {k: v for k, v in profile_indicators.items() if v > 5 and v <= 7}
        
        psychological_indicators['profile'] = {
            'dominant_traits': dominant_traits,
            'secondary_traits': secondary_traits,
            'all_indicators': profile_indicators
        }
        
        return psychological_indicators
    
    def generate_analysis_summary(self, conversation, conversation_context):
        """Gera um resumo da análise da conversa"""
        # Filtrar mensagens do perfil alvo
        target_messages = [msg for msg in conversation if msg['is_target']]
        
        if not target_messages:
            return {"error": "Nenhuma mensagem do perfil alvo encontrada"}
        
        # Análise de atitudes
        attitudes = [msg['analysis']['attitude']['primary_attitude'] 
                    for msg in target_messages 
                    if msg['analysis'].get('attitude', {}).get('primary_attitude')]
        
        attitude_counter = Counter(attitudes)
        total_attitudes = sum(attitude_counter.values())
        attitude_distribution = {attitude: count/total_attitudes 
                               for attitude, count in attitude_counter.items()}
        
        # Análise de emoções (para mensagens de áudio)
        emotions = [msg['analysis']['emotion']['primary_emotion'] 
                   for msg in target_messages 
                   if msg['analysis'].get('emotion', {}).get('primary_emotion')]
        
        emotion_counter = Counter(emotions)
        total_emotions = sum(emotion_counter.values())
        emotion_distribution = {emotion: count/total_emotions 
                              for emotion, count in emotion_counter.items()} if total_emotions else {}
        
        # Análise de temas
        all_themes = []
        for msg in target_messages:
            themes = msg['analysis'].get('themes', [])
            all_themes.extend(themes)
        
        theme_counter = Counter(all_themes)
        
        # Análise comportamental
        behavioral_patterns = [msg['analysis']['behavioral_indicators']['dominant_pattern']['pattern'] 
                              for msg in target_messages 
                              if msg['analysis'].get('behavioral_indicators', {}).get('dominant_pattern', {}).get('pattern')]
        
        behavior_counter = Counter(behavioral_patterns)
        total_behaviors = sum(behavior_counter.values())
        behavior_distribution = {behavior: count/total_behaviors 
                               for behavior, count in behavior_counter.items()} if total_behaviors else {}
        
        # Análise psicológica
        psychological_indicators = {}
        
        # Agregar indicadores psicológicos
        for msg in target_messages:
            psych = msg['analysis'].get('psychological_indicators', {})
            
            for indicator in ['control_need', 'approval_need', 'uncertainty_tolerance', 
                             'emotional_openness', 'cognitive_flexibility', 'autonomy', 'vulnerability']:
                if indicator in psych and 'score' in psych[indicator]:
                    if indicator not in psychological_indicators:
                        psychological_indicators[indicator] = []
                    
                    psychological_indicators[indicator].append(psych[indicator]['score'])
        
        # Calcular médias dos indicadores psicológicos
        psychological_averages = {}
        for indicator, scores in psychological_indicators.items():
            if scores:
                avg_score = sum(scores) / len(scores)
                level = 'alta' if avg_score > 7 else ('média' if avg_score > 4 else 'baixa')
                
                psychological_averages[indicator] = {
                    'average_score': round(avg_score, 2),
                    'level': level
                }
        
        # Análise linguística
        linguistic_averages = {
            'formality': [],
            'complexity': [],
            'hedging': []
        }
        
        for msg in target_messages:
            ling = msg['analysis'].get('linguistic_analysis', {}).get('style_metrics', {})
            
            for metric in linguistic_averages.keys():
                if metric in ling:
                    linguistic_averages[metric].append(ling[metric])
        
        for metric, values in linguistic_averages.items():
            if values:
                linguistic_averages[metric] = sum(values) / len(values)
            else:
                linguistic_averages[metric] = None
        
        # Análise de resposta
        response_times = []
        
        for msg in target_messages:
            if msg['context'].get('conversation_state', {}).get('is_response') and msg['context']['conversation_state'].get('response_time'):
                response_times.append(msg['context']['conversation_state']['response_time'])
        
        avg_response_time = sum(response_times) / len(response_times) if response_times else None
        
        # Construir resumo da análise
        summary = {
            'overview': {
                'total_messages': len(conversation),
                'target_messages': len(target_messages),
                'audio_messages': sum(1 for msg in target_messages if msg['content_type'] == 'audio'),
                'time_range': conversation_context.get('temporal_window', {})
            },
            'communication_patterns': {
                'attitude_distribution': attitude_distribution,
                'primary_attitude': attitude_counter.most_common(1)[0][0] if attitude_counter else None,
                'avg_response_time': avg_response_time,
                'linguistic_style': {
                    'avg_formality': linguistic_averages['formality'],
                    'avg_complexity': linguistic_averages['complexity'],
                    'avg_hedging': linguistic_averages['hedging']
                }
            },
            'content_analysis': {
                'main_themes': theme_counter.most_common(5),
                'emotion_distribution': emotion_distribution,
                'primary_emotion': emotion_counter.most_common(1)[0][0] if emotion_counter else None
            },
            'behavioral_analysis': {
                'behavior_distribution': behavior_distribution,
                'primary_behavior': behavior_counter.most_common(1)[0][0] if behavior_counter else None
            },
            'psychological_profile': {
                'indicators': psychological_averages,
                'dominant_traits': {k: v for k, v in psychological_averages.items() 
                                  if v.get('average_score', 0) > 7}
            }
        }
        
        # Gerar recomendações
        recommendations = self.generate_communication_recommendations(summary, target_messages)
        summary['recommendations'] = recommendations
        
        return summary
    
    def generate_communication_recommendations(self, analysis_summary, messages):
        """Gera recomendações para comunicação eficaz baseadas na análise"""
        recommendations = {
            'general': [],
            'by_topic': {},
            'timing': {},
            'approach': {},
            'wording': {
                'effective': [],
                'avoid': []
            }
        }
        
        # Extrair dados relevantes do resumo
        attitude_dist = analysis_summary['communication_patterns'].get('attitude_distribution', {})
        primary_attitude = analysis_summary['communication_patterns'].get('primary_attitude')
        
        behavior_dist = analysis_summary['behavioral_analysis'].get('behavior_distribution', {})
        primary_behavior = analysis_summary['behavioral_analysis'].get('primary_behavior')
        
        psych_indicators = analysis_summary['psychological_profile'].get('indicators', {})
        
        # 1. Recomendações gerais baseadas na atitude predominante
        if primary_attitude == 'evasiva':
            recommendations['general'].append(
                "Use perguntas diretas que exijam respostas específicas, evitando as que podem ser respondidas com 'sim' ou 'não'."
            )
            recommendations['general'].append(
                "Estabeleça prazos claros e solicite confirmações explícitas para reduzir a ambiguidade."
            )
        
        elif primary_attitude == 'submissa':
            recommendations['general'].append(
                "Ofereça opções claras para escolha, em vez de deixar decisões totalmente abertas."
            )
            recommendations['general'].append(
                "Evite pressão excessiva, pois pode aumentar a submissão por ansiedade em vez de cooperação genuína."
            )
        
        elif primary_attitude == 'colaborativa':
            recommendations['general'].append(
                "Reconheça e agradeça a colaboração proativamente para reforçar este comportamento positivo."
            )
            recommendations['general'].append(
                "Use abordagens de parceria, com decisões compartilhadas para manter o engajamento."
            )
        
        elif primary_attitude == 'resistente':
            recommendations['general'].append(
                "Apresente benefícios pessoais claros antes de fazer solicitações."
            )
            recommendations['general'].append(
                "Evite confrontos diretos; use técnicas de 'sim encadeado' (obter concordância em pontos menores primeiro)."
            )
        
        elif primary_attitude == 'autoritária':
            recommendations['general'].append(
                "Apele à expertise e autoridade da pessoa, enquadrando sugestões como oportunidades para demonstrar competência."
            )
            recommendations['general'].append(
                "Ofereça escolhas que preservem a sensação de controle e autoridade."
            )
        
        # 2. Recomendações baseadas em comportamentos predominantes
        if primary_behavior == 'avoidance':
            recommendations['approach']['structure'] = "Divida solicitações complexas em passos menores e mais gerenciáveis."
            recommendations['approach']['follow_up'] = "Faça acompanhamentos estruturados com perguntas específicas sobre cada etapa acordada."
        
        elif primary_behavior == 'approval_seeking':
            recommendations['approach']['structure'] = "Ofereça reconhecimento frequente e específico para cada tarefa concluída."
            recommendations['approach']['reassurance'] = "Forneça garantias de que as decisões tomadas são apropriadas e valorizadas."
        
        elif primary_behavior == 'control':
            recommendations['approach']['structure'] = "Apresente opções pré-selecionadas em vez de perguntas abertas."
            recommendations['approach']['framing'] = "Enquadre solicitações como oportunidades para demonstrar autoridade e competência."
        
        elif primary_behavior == 'emotional':
            recommendations['approach']['engagement'] = "Comece interações com conexão pessoal antes de tratar de assuntos profissionais."
            recommendations['approach']['acknowledgment'] = "Reconheça explicitamente os sentimentos expressos antes de abordar questões práticas."
        
        # 3. Recomendações sobre palavras específicas a usar ou evitar
        control_need = psych_indicators.get('control_need', {}).get('level')
        approval_need = psych_indicators.get('approval_need', {}).get('level')
        uncertainty_tolerance = psych_indicators.get('uncertainty_tolerance', {}).get('level')
        
        # Palavras eficazes com base em indicadores psicológicos
        if control_need == 'alta':
            recommendations['wording']['effective'].extend([
                "você decide", "sob seu controle", "sua autoridade", "sua liderança",
                "você determina"
            ])
            recommendations['wording']['avoid'].extend([
                "você deve", "é necessário que", "obrigatório", "tem que"
            ])
        
        if approval_need == 'alta':
            recommendations['wording']['effective'].extend([
                "excelente trabalho", "muito competente", "impressionante", "valorizo muito",
                "confio em você"
            ])
        
        if uncertainty_tolerance == 'baixa':
            recommendations['wording']['effective'].extend([
                "plano detalhado", "cronograma definido", "estrutura clara", 
                "passos específicos", "procedimento estabelecido"
            ])
            recommendations['wording']['avoid'].extend([
                "vamos ver", "talvez", "possivelmente", "ainda não sabemos",
                "em aberto"
            ])
        
        # 4. Recomendações de timing
        # Analisar tempos de resposta por hora do dia
        hour_responses = defaultdict(list)
        
        for msg in messages:
            if msg['context'].get('conversation_state', {}).get('is_response') and msg['context']['conversation_state'].get('response_time'):
                dt = datetime.fromisoformat(msg['datetime'])
                hour = dt.hour
                hour_responses[hour].append(msg['context']['conversation_state']['response_time'])
        
        # Encontrar melhores horários (com respostas mais rápidas)
        best_hours = []
        if hour_responses:
            avg_by_hour = {hour: sum(times)/len(times) for hour, times in hour_responses.items() if len(times) >= 2}
            sorted_hours = sorted(avg_by_hour.items(), key=lambda x: x[1])[:3]  # 3 melhores horas
            best_hours = [hour for hour, _ in sorted_hours]
        
        if best_hours:
            recommendations['timing']['best_hours'] = best_hours
            recommendations['general'].append(
                f"Envie mensagens importantes entre {', '.join([f'{h}h' for h in best_hours])} para obter respostas mais rápidas."
            )
        
        # 5. Recomendações por tema
        # Analisar atitudes por tema
        theme_attitudes = defaultdict(list)
        
        for msg in messages:
            themes = msg['analysis'].get('themes', [])
            attitude = msg['analysis'].get('attitude', {}).get('primary_attitude')
            
            if themes and attitude:
                for theme in themes:
                    theme_clean = theme.replace('#', '')
                    theme_attitudes[theme_clean].append(attitude)
        
        # Gerar recomendações específicas por tema
        for theme, attitudes in theme_attitudes.items():
            counter = Counter(attitudes)
            most_common = counter.most_common(1)[0][0]
            
            if theme == 'plantao' and most_common == 'evasiva':
                recommendations['by_topic'][theme] = "Para assuntos de plantão, use mensagens curtas e diretas com perguntas específicas que exijam respostas concretas."
            elif theme == 'feriado' and most_common == 'resistente':
                recommendations['by_topic'][theme] = "Inicie discussões sobre feriados com pelo menos duas semanas de antecedência, apresentando opções predefinidas."
            elif theme == 'pagamento' and most_common in ['resistente', 'evasiva']:
                recommendations['by_topic'][theme] = "Para questões de pagamento, faça referência a procedimentos oficiais ou prazos estabelecidos por terceiros."
            elif theme == 'reuniao' and most_common == 'colaborativa':
                recommendations['by_topic'][theme] = "Ao agendar reuniões, mantenha o tom colaborativo atual que tem gerado boas respostas."
        
        # Converter lista de palavras para formato mais legível
        if recommendations['wording']['effective']:
            effective_phrases = recommendations['wording']['effective']
            recommendations['wording']['effective'] = [
                f"Use termos como: {', '.join(effective_phrases[0:min(5, len(effective_phrases))])}"
            ]
        
        if recommendations['wording']['avoid']:
            avoid_phrases = recommendations['wording']['avoid']
            recommendations['wording']['avoid'] = [
                f"Evite termos como: {', '.join(avoid_phrases[0:min(5, len(avoid_phrases))])}"
            ]
        
        return recommendations

    def save_analysis_report(self, profile_id, conversation, analysis_summary):
        """Salva um relatório de análise para um perfil"""
        if profile_id not in self.profiles:
            logger.error(f"Perfil não encontrado: {profile_id}")
            return False
        
        try:
            # Criar ID único para a análise
            analysis_id = f"analysis_{datetime.now().strftime('%Y%m%d%H%M%S')}"
            
            # Preparar estrutura de análise
            analysis_data = {
                "id": analysis_id,
                "timestamp": datetime.now().isoformat(),
                "summary": analysis_summary,
                "conversation_meta": {
                    "message_count": len(conversation),
                    "target_message_count": sum(1 for msg in conversation if msg.get('is_target', False)),
                    "first_message": conversation[0]['datetime'] if conversation else None,
                    "last_message": conversation[-1]['datetime'] if conversation else None
                }
            }
            
            # Adicionar a análise ao perfil
            self.profiles[profile_id]['analyses'].append(analysis_data)
            
            # Atualizar padrões comportamentais do perfil com base na análise
            self.update_profile_patterns(profile_id, analysis_summary)
            
            # Salvar conversas analisadas em arquivo separado para não sobrecarregar o perfil
            conversations_dir = os.path.join(self.config["DATA_DIR"], profile_id, "conversations")
            os.makedirs(conversations_dir, exist_ok=True)
            
            conversation_path = os.path.join(conversations_dir, f"{analysis_id}.json")
            
            with open(conversation_path, 'w', encoding='utf-8') as f:
                json.dump(conversation, f, ensure_ascii=False, indent=2)
            
            # Atualizar timestamp do perfil
            self.profiles[profile_id]['last_updated'] = datetime.now().isoformat()
            
            # Salvar perfil atualizado
            profile_path = os.path.join(self.config["PROFILES_DIR"], f"{profile_id}.json")
            
            with open(profile_path, 'w', encoding='utf-8') as f:
                json.dump(self.profiles[profile_id], f, ensure_ascii=False, indent=2)
            
            logger.info(f"Análise {analysis_id} salva para o perfil {profile_id}")
            return analysis_id
            
        except Exception as e:
            logger.error(f"Erro ao salvar análise para perfil {profile_id}: {e}")
            return False
    
    def update_profile_patterns(self, profile_id, analysis_summary):
        """Atualiza os padrões comportamentais e psicológicos do perfil com base na análise"""
        if profile_id not in self.profiles:
            logger.error(f"Perfil não encontrado: {profile_id}")
            return False
        
        try:
            profile = self.profiles[profile_id]
            
            # Atualizar padrões comportamentais
            behavior_dist = analysis_summary.get('behavioral_analysis', {}).get('behavior_distribution', {})
            
            if behavior_dist:
                if 'behavioral_patterns' not in profile:
                    profile['behavioral_patterns'] = {}
                
                if 'communication_style' not in profile['behavioral_patterns']:
                    profile['behavioral_patterns']['communication_style'] = {}
                
                # Mesclar distribuições novas com existentes
                for behavior, score in behavior_dist.items():
                    if behavior in profile['behavioral_patterns']['communication_style']:
                        # Média ponderada (75% valor anterior, 25% novo valor)
                        profile['behavioral_patterns']['communication_style'][behavior] = (
                            profile['behavioral_patterns']['communication_style'][behavior] * 0.75 + score * 0.25
                        )
                    else:
                        profile['behavioral_patterns']['communication_style'][behavior] = score
            
            # Atualizar padrões psicológicos
            psych_indicators = analysis_summary.get('psychological_profile', {}).get('indicators', {})
            
            if psych_indicators:
                if 'psychological_profile' not in profile:
                    profile['psychological_profile'] = {}
                
                # Atualizar cada indicador
                for indicator, data in psych_indicators.items():
                    category = None
                    
                    if indicator in ['control_need', 'approval_need']:
                        category = 'motivation_factors'
                    elif indicator in ['uncertainty_tolerance', 'cognitive_flexibility']:
                        category = 'cognitive_patterns'
                    elif indicator in ['emotional_openness', 'vulnerability']:
                        category = 'emotional_tendencies'
                    elif indicator in ['autonomy']:
                        category = 'personality_traits'
                    
                    if category:
                        if category not in profile['psychological_profile']:
                            profile['psychological_profile'][category] = {}
                        
                        # Atualizar ou adicionar o indicador
                        if indicator in profile['psychological_profile'][category]:
                            # Média ponderada para atualização incremental
                            prev_score = profile['psychological_profile'][category][indicator]['score']
                            new_score = data.get('average_score', 5)
                            updated_score = prev_score * 0.75 + new_score * 0.25
                            
                            profile['psychological_profile'][category][indicator] = {
                                'score': updated_score,
                                'level': data.get('level', 'média')
                            }
                        else:
                            profile['psychological_profile'][category][indicator] = {
                                'score': data.get('average_score', 5),
                                'level': data.get('level', 'média')
                            }
            
            # Atualizar estatísticas de comunicação
            comm_patterns = analysis_summary.get('communication_patterns', {})
            
            if comm_patterns:
                if 'communication_stats' not in profile:
                    profile['communication_stats'] = {}
                
                # Atualizar distribuição de atitudes
                if 'attitude_distribution' in comm_patterns:
                    if 'attitude_distribution' not in profile['communication_stats']:
                        profile['communication_stats']['attitude_distribution'] = {}
                    
                    for attitude, score in comm_patterns['attitude_distribution'].items():
                        if attitude in profile['communication_stats']['attitude_distribution']:
                            # Média ponderada
                            profile['communication_stats']['attitude_distribution'][attitude] = (
                                profile['communication_stats']['attitude_distribution'][attitude] * 0.75 + score * 0.25
                            )
                        else:
                            profile['communication_stats']['attitude_distribution'][attitude] = score
                
                # Atualizar tempo médio de resposta
                if 'avg_response_time' in comm_patterns and comm_patterns['avg_response_time'] is not None:
                    new_time = comm_patterns['avg_response_time']
                    
                    if 'response_times' not in profile['communication_stats']:
                        profile['communication_stats']['response_times'] = []
                    
                    profile['communication_stats']['response_times'].append(new_time)
                    
                    # Manter apenas os 10 tempos mais recentes
                    if len(profile['communication_stats']['response_times']) > 10:
                        profile['communication_stats']['response_times'] = profile['communication_stats']['response_times'][-10:]
                    
                    profile['communication_stats']['avg_response_time'] = sum(profile['communication_stats']['response_times']) / len(profile['communication_stats']['response_times'])
            
            # Atualizar recomendações
            if 'recommendations' in analysis_summary:
                if 'recommendations' not in profile:
                    profile['recommendations'] = {
                        'general': [],
                        'by_topic': {},
                        'by_context': {}
                    }
                
                # Atualizar recomendações gerais (manter apenas as 5 mais recentes)
                if 'general' in analysis_summary['recommendations']:
                    new_recs = analysis_summary['recommendations']['general']
                    current_recs = profile['recommendations']['general']
                    
                    # Adicionar novas recomendações únicas
                    for rec in new_recs:
                        if rec not in current_recs:
                            current_recs.append(rec)
                    
                    # Manter apenas as 7 mais recentes
                    profile['recommendations']['general'] = current_recs[-7:] if len(current_recs) > 7 else current_recs
                
                # Atualizar recomendações por tópico
                if 'by_topic' in analysis_summary['recommendations']:
                    for topic, rec in analysis_summary['recommendations']['by_topic'].items():
                        profile['recommendations']['by_topic'][topic] = rec
            
            logger.info(f"Padrões do perfil {profile_id} atualizados com nova análise")
            return True
            
        except Exception as e:
            logger.error(f"Erro ao atualizar padrões do perfil {profile_id}: {e}")
            return False
    
    def generate_text_report(self, profile_id, analysis_id=None):
        """Gera um relatório em formato texto para uma análise específica"""
        if profile_id not in self.profiles:
            logger.error(f"Perfil não encontrado: {profile_id}")
            return None
        
        try:
            profile = self.profiles[profile_id]
            
            # Se não for especificada uma análise, usar a mais recente
            if not analysis_id and profile.get('analyses'):
                analysis = profile['analyses'][-1]
            elif analysis_id:
                analysis = next((a for a in profile['analyses'] if a['id'] == analysis_id), None)
                if not analysis:
                    logger.error(f"Análise {analysis_id} não encontrada para o perfil {profile_id}")
                    return None
            else:
                logger.error(f"Nenhuma análise disponível para o perfil {profile_id}")
                return None
            
            # Carregar conversa associada à análise
            conversation_path = os.path.join(
                self.config["DATA_DIR"], 
                profile_id, 
                "conversations", 
                f"{analysis['id']}.json"
            )
            
            conversation = []
            if os.path.exists(conversation_path):
                with open(conversation_path, 'r', encoding='utf-8') as f:
                    conversation = json.load(f)
            
            # Gerar nome de arquivo para o relatório
            report_filename = f"{profile_id}_{analysis['id']}_report.txt"
            report_path = os.path.join(self.config["EXPORT_DIR"], report_filename)
            
            # Criar diretório de exportação se não existir
            os.makedirs(os.path.dirname(report_path), exist_ok=True)
            
            # Gerar relatório
            with open(report_path, 'w', encoding='utf-8') as f:
                # Cabeçalho
                f.write("=" * 80 + "\n")
                f.write(f"RELATÓRIO DE ANÁLISE COMPORTAMENTAL - {profile['name'].upper()}\n")
                f.write(f"Gerado em: {datetime.now().strftime('%d/%m/%Y %H:%M:%S')}\n")
                f.write("=" * 80 + "\n\n")
                
                # Informações do perfil
                f.write("INFORMAÇÕES DO PERFIL\n")
                f.write("-" * 80 + "\n")
                f.write(f"Nome: {profile['name']}\n")
                if profile.get('description'):
                    f.write(f"Descrição: {profile['description']}\n")
                
                created_at = datetime.fromisoformat(profile['created_at']).strftime('%d/%m/%Y')
                f.write(f"Perfil criado em: {created_at}\n")
                
                analysis_time = datetime.fromisoformat(analysis['timestamp']).strftime('%d/%m/%Y %H:%M')
                f.write(f"Data desta análise: {analysis_time}\n")
                
                msg_count = analysis['conversation_meta']['message_count']
                target_count = analysis['conversation_meta']['target_message_count']
                f.write(f"Mensagens analisadas: {msg_count} (sendo {target_count} de {profile['name']})\n\n")
                
                # Análise de mensagens
                if conversation:
                    f.write("ANÁLISE DE MENSAGENS\n")
                    f.write("-" * 80 + "\n\n")
                    
                    for msg in conversation:
                        # Pular mensagens que não são do alvo se a conversa for muito longa
                        if len(conversation) > 30 and not msg.get('is_target'):
                            continue
                        
                        # Formatar data e hora
                        dt = datetime.fromisoformat(msg['datetime'])
                        dt_str = dt.strftime('%d/%m/%Y %H:%M')
                        
                        # Linha básica da mensagem
                        is_target = msg.get('is_target', False)
                        author_prefix = "➤ " if is_target else ""
                        f.write(f"[{dt_str}] {author_prefix}{msg['author']}: ")
                        
                        # Conteúdo da mensagem
                        if msg['content_type'] == 'audio':
                            f.write(f"{msg.get('media_file', '[ÁUDIO]')}\n")
                            if msg.get('transcription'):
                                f.write(f"📝 Transcrição: \"{msg['transcription']}\"\n")
                            
                            # Análise emocional
                            if is_target and msg.get('analysis', {}).get('emotion'):
                                emotion = msg['analysis']['emotion']
                                emotion_text = f"🎭 Emoção: {emotion.get('emotional_blend', '')}"
                                if 'intensity' in emotion:
                                    emotion_text += f" (intensidade: {emotion['intensity']}/10)"
                                if 'authenticity' in emotion:
                                    emotion_text += f", autenticidade: {emotion['authenticity']}%"
                                f.write(f"{emotion_text}\n")
                        else:
                            f.write(f"{msg['content']}\n")
                        
                        # Adicionar análises para mensagens do alvo
                        if is_target:
                            # Atitude
                            if msg.get('analysis', {}).get('attitude'):
                                attitude = msg['analysis']['attitude']
                                f.write(f"🎯 Atitude: {attitude['primary_attitude']} (intensidade: {attitude['strength']})\n")
                            
                            # Temas
                            if msg.get('analysis', {}).get('themes'):
                                themes = msg['analysis']['themes']
                                f.write(f"🏷️ Temas: {', '.join(themes)}\n")
                            
                            # Comportamento
                            if msg.get('analysis', {}).get('behavioral_indicators', {}).get('dominant_pattern'):
                                behavior = msg['analysis']['behavioral_indicators']['dominant_pattern']
                                if behavior['pattern'] != 'neutral':
                                    f.write(f"🔍 Comportamento: {behavior['pattern']} (score: {behavior['score']})\n")
                                    
                                    # Adicionar padrões secundários se existirem
                                    if behavior.get('secondary_patterns'):
                                        f.write(f"   Padrões secundários: {', '.join(behavior['secondary_patterns'])}\n")
                            
                            # Indicadores psicológicos notáveis (alto ou baixo)
                            notable_indicators = []
                            psych = msg.get('analysis', {}).get('psychological_indicators', {})
                            
                            for indicator in ['control_need', 'approval_need', 'vulnerability']:
                                if indicator in psych and psych[indicator].get('level') == 'alta':
                                    notable_indicators.append(f"{indicator}: {psych[indicator]['level']}")
                            
                            if notable_indicators:
                                f.write(f"💡 Indicadores: {', '.join(notable_indicators)}\n")
                        
                        f.write("\n")  # Linha em branco entre mensagens
                
                # Resumo da análise
                summary = analysis['summary']
                
                f.write("\nRESUMO DA ANÁLISE\n")
                f.write("-" * 80 + "\n")
                
                # Padrões de comunicação
                comm_patterns = summary.get('communication_patterns', {})
                f.write("📊 Padrões de Comunicação:\n")
                
                if comm_patterns.get('primary_attitude'):
                    f.write(f"- Atitude predominante: {comm_patterns['primary_attitude']}\n")
                
                if comm_patterns.get('attitude_distribution'):
                    f.write("- Distribuição de atitudes:\n")
                    for attitude, value in comm_patterns['attitude_distribution'].items():
                        f.write(f"  • {attitude}: {value*100:.1f}%\n")
                
                if comm_patterns.get('avg_response_time'):
                    response_time = comm_patterns['avg_response_time']
                    if response_time < 60:
                        response_str = f"{response_time:.1f} minutos"
                    else:
                        hours = int(response_time // 60)
                        mins = int(response_time % 60)
                        response_str = f"{hours}h {mins}min"
                    
                    f.write(f"- Tempo médio de resposta: {response_str}\n")
                
                # Conteúdo e emoções
                content = summary.get('content_analysis', {})
                f.write("\n📋 Análise de Conteúdo:\n")
                
                if content.get('main_themes'):
                    f.write("- Temas principais:\n")
                    for theme, count in content['main_themes']:
                        f.write(f"  • {theme}: {count} ocorrências\n")
                
                if content.get('emotion_distribution'):
                    f.write("- Emoções detectadas:\n")
                    for emotion, value in content['emotion_distribution'].items():
                        f.write(f"  • {emotion}: {value*100:.1f}%\n")
                
                # Análise comportamental
                behavior = summary.get('behavioral_analysis', {})
                f.write("\n🔍 Análise Comportamental:\n")
                
                if behavior.get('primary_behavior'):
                    f.write(f"- Padrão comportamental predominante: {behavior['primary_behavior']}\n")
                
                if behavior.get('behavior_distribution'):
                    f.write("- Distribuição de comportamentos:\n")
                    for pattern, value in behavior['behavior_distribution'].items():
                        f.write(f"  • {pattern}: {value*100:.1f}%\n")
                
                # Perfil psicológico
                psych = summary.get('psychological_profile', {})
                f.write("\n🧠 Perfil Psicológico:\n")
                
                if psych.get('indicators'):
                    f.write("- Indicadores psicológicos:\n")
                    for indicator, data in psych['indicators'].items():
                        indicator_name = {
                            'control_need': 'Necessidade de controle',
                            'approval_need': 'Necessidade de aprovação',
                            'uncertainty_tolerance': 'Tolerância a incertezas',
                            'emotional_openness': 'Abertura emocional',
                            'cognitive_flexibility': 'Flexibilidade cognitiva',
                            'autonomy': 'Autonomia',
                            'vulnerability': 'Vulnerabilidade'
                        }.get(indicator, indicator)
                        
                        f.write(f"  • {indicator_name}: {data['level']} ({data['average_score']}/10)\n")
                
                # Recomendações
                recommendations = summary.get('recommendations', {})
                f.write("\nRECOMENDAÇÕES ESTRATÉGICAS\n")
                f.write("-" * 80 + "\n")
                
                # Recomendações gerais
                if recommendations.get('general'):
                    f.write("🧠 Recomendações Gerais:\n")
                    for i, rec in enumerate(recommendations['general'], 1):
                        f.write(f"{i}. {rec}\n")
                    f.write("\n")
                
                # Recomendações por tema
                if recommendations.get('by_topic'):
                    f.write("🏷️ Recomendações por Tema:\n")
                    for theme, rec in recommendations['by_topic'].items():
                        f.write(f"- {theme}: {rec}\n")
                    f.write("\n")
                
                # Recomendações de timing
                if recommendations.get('timing', {}).get('best_hours'):
                    best_hours = recommendations['timing']['best_hours']
                    f.write(f"⏰ Melhores horários para comunicação: {', '.join([f'{h}h' for h in best_hours])}\n\n")
                
                # Recomendações de linguagem
                if recommendations.get('wording', {}).get('effective'):
                    f.write("💬 Linguagem Recomendada:\n")
                    for phrase in recommendations['wording']['effective']:
                        f.write(f"- {phrase}\n")
                
                if recommendations.get('wording', {}).get('avoid'):
                    f.write("\n⚠️ Linguagem a Evitar:\n")
                    for phrase in recommendations['wording']['avoid']:
                        f.write(f"- {phrase}\n")
            
            logger.info(f"Relatório gerado: {report_path}")
            return report_path
            
        except Exception as e:
            logger.error(f"Erro ao gerar relatório para perfil {profile_id}: {e}")
            return None

    def update_profile_list(self):
        """Atualiza a lista de perfis na interface"""
        self.profile_listbox.delete(0, tk.END)
        
        if not self.profiles:
            self.profile_listbox.insert(tk.END, "Nenhum perfil disponível")
            return
        
        # Ordenar perfis por nome
        sorted_profiles = sorted(self.profiles.items(), key=lambda x: x[1]['name'])
        
        for profile_id, profile_data in sorted_profiles:
            self.profile_listbox.insert(tk.END, profile_data['name'])
            
            # Associar o ID do perfil ao item da lista
            index = self.profile_listbox.size() - 1
            self.profile_listbox.itemconfig(index, {'profile_id': profile_id})
    
    def update_recent_analyses(self):
        """Atualiza a lista de análises recentes"""
        self.recent_analyses_listbox.delete(0, tk.END)
        
        if not self.current_profile or self.current_profile not in self.profiles:
            self.recent_analyses_listbox.insert(tk.END, "Selecione um perfil")
            return
        
        profile = self.profiles[self.current_profile]
        
        if not profile.get('analyses'):
            self.recent_analyses_listbox.insert(tk.END, "Nenhuma análise disponível")
            return
        
        # Mostrar as análises mais recentes primeiro
        analyses = sorted(profile['analyses'], key=lambda x: x['timestamp'], reverse=True)
        
        for analysis in analyses[:10]:  # Limitar a 10 análises recentes
            timestamp = datetime.fromisoformat(analysis['timestamp']).strftime('%d/%m/%Y %H:%M')
            self.recent_analyses_listbox.insert(tk.END, f"{timestamp} - {analysis['id']}")
            
            # Associar o ID da análise ao item da lista
            index = self.recent_analyses_listbox.size() - 1
            self.recent_analyses_listbox.itemconfig(index, {'analysis_id': analysis['id']})
    
    def update_system_stats(self):
        """Atualiza as estatísticas do sistema na tela inicial"""
        try:
            self.system_stats_text.config(state=tk.NORMAL)
            self.system_stats_text.delete(1.0, tk.END)
            
            # Contar estatísticas
            profile_count = len(self.profiles)
            
            analysis_count = 0
            for profile in self.profiles.values():
                analysis_count += len(profile.get('analyses', []))
            
            # Estatísticas de armazenamento
            profiles_size = self.get_directory_size(self.config["PROFILES_DIR"]) / (1024 * 1024)  # MB
            data_size = self.get_directory_size(self.config["DATA_DIR"]) / (1024 * 1024)  # MB
            
            stats_text = f"Total de perfis: {profile_count}\n"
            stats_text += f"Total de análises: {analysis_count}\n"
            stats_text += f"Tamanho dos dados: {data_size:.2f} MB\n"
            stats_text += f"Último acesso: {datetime.now().strftime('%d/%m/%Y %H:%M:%S')}\n"
            
            self.system_stats_text.insert(tk.END, stats_text)
            self.system_stats_text.config(state=tk.DISABLED)
            
        except Exception as e:
            logger.error(f"Erro ao atualizar estatísticas do sistema: {e}")
    
    def get_directory_size(self, path):
        """Calcula o tamanho total de um diretório em bytes"""
        total_size = 0
        
        if not os.path.exists(path):
            return 0
        
        for dirpath, dirnames, filenames in os.walk(path):
            for filename in filenames:
                file_path = os.path.join(dirpath, filename)
                if os.path.exists(file_path):
                    total_size += os.path.getsize(file_path)
        
        return total_size
    
    def on_profile_selected(self, event):
        """Manipula a seleção de um perfil na lista"""
        selection = self.profile_listbox.curselection()
        
        if not selection:
            return
        
        # Obter o índice e o item selecionado
        index = selection[0]
        
        # Verificar se há perfis disponíveis
        if self.profiles and self.profile_listbox.get(index) != "Nenhum perfil disponível":
            # Obter o ID do perfil associado ao item
            profile_id = self.profile_listbox.itemcget(index, 'profile_id')
            
            if profile_id:
                # Definir o perfil atual
                self.current_profile = profile_id
                
                # Atualizar interface
                self.update_recent_analyses()
                self.display_profile_info()
                self.update_recommendations_tab()
                
                # Mudar para a aba de perfil
                self.notebook.select(self.profile_tab)
    
    def on_analysis_selected(self, event):
        """Manipula a seleção de uma análise na lista"""
        selection = self.recent_analyses_listbox.curselection()
        
        if not selection:
            return
        
        # Obter o índice e o item selecionado
        index = selection[0]
        item_text = self.recent_analyses_listbox.get(index)
        
        # Verificar se há análises disponíveis
        if item_text != "Nenhuma análise disponível" and item_text != "Selecione um perfil":
            # Obter o ID da análise associado ao item
            analysis_id = self.recent_analyses_listbox.itemcget(index, 'analysis_id')
            
            if analysis_id:
                # Definir a análise atual
                self.current_analysis = analysis_id
                
                # Atualizar interface
                self.display_analysis_info()
                self.display_conversation()
                self.update_recommendations_tab()
                
                # Mudar para a aba de análise
                self.notebook.select(self.analysis_tab)

    def perform_analysis(self, options):
        """Executa a análise com as opções selecionadas"""
        try:
            # Mostrar progresso
            self.show_progress("Iniciando análise...")
            
            # 1. Carregar dados
            if options['source_type'] == "whatsapp":
                self.update_progress(0.1, "Carregando conversa do WhatsApp...")
                conversation = self.load_whatsapp_chat(options['path'])
            else:
                # Implementação para pasta com arquivos de mídia
                self.update_progress(0.1, "Carregando arquivos da pasta...")
                # TODO: Implementar carregamento de arquivos de pasta
                conversation = []
            
            if not conversation:
                self.hide_progress()
                messagebox.showerror("Erro", "Não foi possível carregar os dados para análise")
                return
            
            # 2. Processar arquivos de mídia
            if options['process_audio']:
                self.update_progress(0.2, "Processando arquivos de mídia...")
                
                # Diretório fonte dos arquivos de mídia
                if options['source_type'] == "whatsapp":
                    media_dir = os.path.dirname(options['path'])
                else:
                    media_dir = options['path']
                
                # Função de callback para progresso
                def media_progress(progress):
                    self.update_progress(0.2 + progress * 0.3, f"Processando mídia... {int(progress * 100)}%")
                
                conversation = self.process_media_files(conversation, media_dir, media_progress)
            
            # 3. Analisar conversa
            self.update_progress(0.5, "Analisando conversa...")
            
            # Função de callback para progresso
            def analysis_progress(progress):
                self.update_progress(0.5 + progress * 0.4, f"Analisando... {int(progress * 100)}%")
            
            conversation, analysis_summary = self.analyze_conversation(conversation, analysis_progress)
            
            # 4. Salvar análise
            self.update_progress(0.9, "Salvando análise...")
            analysis_id = self.save_analysis_report(self.current_profile, conversation, analysis_summary)
            
            if not analysis_id:
                self.hide_progress()
                messagebox.showerror("Erro", "Não foi possível salvar a análise")
                return
            
            # 5. Atualizar interface
            self.update_progress(0.95, "Atualizando interface...")
            self.current_analysis = analysis_id
            self.update_recent_analyses()
            
            # 6. Gerar relatório
            self.update_progress(0.98, "Gerando relatório...")
            self.generate_text_report(self.current_profile, analysis_id)
            
            # Atualizar exibição
            self.display_analysis_info()
            self.display_conversation()
            self.update_recommendations_tab()
            
            # Esconder progresso
            self.hide_progress()
            
            # Mostrar mensagem de sucesso
            messagebox.showinfo("Sucesso", "Análise concluída com sucesso!")
            
            # Mudar para a aba de análise
            self.notebook.select(self.analysis_tab)
            
        except Exception as e:
            self.hide_progress()
            logger.error(f"Erro durante a análise: {e}")
            messagebox.showerror("Erro", f"Ocorreu um erro durante a análise: {str(e)}")

    def display_conversation(self):
        """Exibe a conversa analisada"""
        if (not self.current_profile or self.current_profile not in self.profiles or 
            not self.current_analysis):
            return
        
        try:
            # Limpar conteúdo anterior
            for widget in self.conversation_content_frame.winfo_children():
                widget.destroy()
            
            # Caminho para o arquivo da conversa
            conversation_path = os.path.join(
                self.config["DATA_DIR"], 
                self.current_profile, 
                "conversations", 
                f"{self.current_analysis}.json"
            )
            
            if not os.path.exists(conversation_path):
                ttk.Label(self.conversation_content_frame, 
                        text="Arquivo de conversa não encontrado", 
                        font=("Arial", 12)).pack(pady=50)
                return
            
            # Carregar conversa
            with open(conversation_path, 'r', encoding='utf-8') as f:
                conversation = json.load(f)
            
            # Container principal
            main_frame = ttk.Frame(self.conversation_content_frame)
            main_frame.pack(fill=tk.BOTH, expand=True)
            
            # Título
            ttk.Label(main_frame, text="Conversa Analisada", font=("Arial", 14, "bold")).pack(anchor=tk.W, pady=(0, 10))
            
            # Container da conversa com scroll
            conversation_canvas = tk.Canvas(main_frame)
            conversation_scrollbar = ttk.Scrollbar(main_frame, orient=tk.VERTICAL, command=conversation_canvas.yview)
            
            conversation_canvas.pack(side=tk.LEFT, fill=tk.BOTH, expand=True)
            conversation_scrollbar.pack(side=tk.RIGHT, fill=tk.Y)
            
            conversation_canvas.configure(yscrollcommand=conversation_scrollbar.set)
            
            # Frame dentro do canvas para conteúdo
            messages_frame = ttk.Frame(conversation_canvas)
            canvas_window = conversation_canvas.create_window((0, 0), window=messages_frame, anchor=tk.NW)
            
            # Exibir mensagens
            for msg in conversation:
                # Criar frame para a mensagem
                is_target = msg.get('is_target', False)
                
                msg_frame = ttk.Frame(messages_frame)
                msg_frame.pack(fill=tk.X, pady=5)
                
                # Frame para cabeçalho da mensagem
                header_frame = ttk.Frame(msg_frame)
                header_frame.pack(fill=tk.X)
                
                # Data/hora e autor
                dt = datetime.fromisoformat(msg['datetime'])
                dt_str = dt.strftime('%d/%m/%Y %H:%M')
                
                author_prefix = "➤ " if is_target else ""
                header = f"[{dt_str}] {author_prefix}{msg['author']}"
                
                header_label = ttk.Label(header_frame, text=header)
                header_label.pack(side=tk.LEFT)
                
                # Conteúdo da mensagem
                content_frame = ttk.Frame(msg_frame)
                content_frame.pack(fill=tk.X, padx=20)
                
                # Exibir conteúdo conforme o tipo
                if msg['content_type'] == 'audio':
                    ttk.Label(content_frame, text=f"[Áudio: {msg.get('media_file', '')}]").pack(anchor=tk.W)
                    
                    if msg.get('transcription'):
                        transcription_text = f"📝: \"{msg['transcription']}\""
                        ttk.Label(content_frame, text=transcription_text, wraplength=500).pack(anchor=tk.W)
                else:
                    ttk.Label(content_frame, text=msg['content'], wraplength=500).pack(anchor=tk.W)
                
                # Exibir análises para mensagens do alvo
                if is_target:
                    analysis_frame = ttk.Frame(msg_frame)
                    analysis_frame.pack(fill=tk.X, padx=40, pady=5)
                    
                    # Mostrar informações da análise
                    analysis_text = ""
                    
                    # Emoção (para áudios)
                    if msg['content_type'] == 'audio' and msg.get('analysis', {}).get('emotion'):
                        emotion = msg['analysis']['emotion']
                        emotion_text = f"🎭 Emoção: {emotion.get('emotional_blend', '')}"
                        if 'intensity' in emotion:
                            emotion_text += f" (intensidade: {emotion['intensity']}/10)"
                        analysis_text += emotion_text + "\n"
                    
                    # Atitude
                    if msg.get('analysis', {}).get('attitude'):
                        attitude = msg['analysis']['attitude']
                        attitude_text = f"🎯 Atitude: {attitude['primary_attitude']} (intensidade: {attitude['strength']})"
                        analysis_text += attitude_text + "\n"
                    
                    # Temas
                    if msg.get('analysis', {}).get('themes'):
                        themes = msg['analysis']['themes']
                        themes_text = f"🏷️ Temas: {', '.join(themes)}"
                        analysis_text += themes_text + "\n"
                    
                    # Comportamento
                    if msg.get('analysis', {}).get('behavioral_indicators', {}).get('dominant_pattern'):
                        behavior = msg['analysis']['behavioral_indicators']['dominant_pattern']
                        if behavior['pattern'] != 'neutral':
                            behavior_text = f"🔍 Comportamento: {behavior['pattern']} (score: {behavior['score']})"
                            analysis_text += behavior_text
                    
                    # Mostrar texto de análise se houver
                    if analysis_text:
                        analysis_label = ttk.Label(analysis_frame, text=analysis_text, foreground="#555555")
                        analysis_label.pack(anchor=tk.W)
            
            # Configurar scroll
            def configure_scroll(event):
                conversation_canvas.configure(scrollregion=conversation_canvas.bbox("all"), width=event.width)
            
            messages_frame.bind("<Configure>", configure_scroll)
            
            # Vincular roda do mouse ao scroll
            def on_mousewheel(event):
                conversation_canvas.yview_scroll(int(-1*(event.delta/120)), "units")
            
            conversation_canvas.bind_all("<MouseWheel>", on_mousewheel)
            
        except Exception as e:
            logger.error(f"Erro ao exibir conversa: {e}")
            # Limpar conteúdo anterior
            for widget in self.conversation_content_frame.winfo_children():
                widget.destroy()
            
            ttk.Label(self.conversation_content_frame, 
                    text=f"Erro ao exibir conversa: {str(e)}", 
                    font=("Arial", 12)).pack(pady=50)

    def generate_report_for_selected(self):
        """Gera um relatório para a análise selecionada"""
        if not self.current_profile or not self.current_analysis:
            messagebox.showinfo("Aviso", "Selecione uma análise para gerar relatório")
            return
        
        report_path = self.generate_text_report(self.current_profile, self.current_analysis)
        
        if report_path:
            response = messagebox.askyesno(
                "Relatório Gerado",
                f"Relatório gerado com sucesso em:\n{report_path}\n\nDeseja abrir o relatório agora?"
            )
            
            if response:
                self.open_file(report_path)
        else:
            messagebox.showerror("Erro", "Não foi possível gerar o relatório")
    
    def export_report_for_analysis(self, profile_id, analysis_id):
        """Exporta o relatório de uma análise específica"""
        report_path = self.generate_text_report(profile_id, analysis_id)
        
        if not report_path:
            messagebox.showerror("Erro", "Não foi possível gerar o relatório")
            return
        
        # Abrir diálogo para escolher local de exportação
        save_path = filedialog.asksaveasfilename(
            title="Exportar Relatório",
            defaultextension=".txt",
            filetypes=[("Arquivo de texto", "*.txt"), ("Todos os arquivos", "*.*")]
        )
        
        if not save_path:
            return
        
        try:
            # Copiar arquivo
            shutil.copy2(report_path, save_path)
            
            response = messagebox.askyesno(
                "Relatório Exportado",
                f"Relatório exportado com sucesso para:\n{save_path}\n\nDeseja abrir o relatório agora?"
            )
            
            if response:
                self.open_file(save_path)
                
        except Exception as e:
            logger.error(f"Erro ao exportar relatório: {e}")
            messagebox.showerror("Erro", f"Não foi possível exportar o relatório: {str(e)}")
    
    def open_file(self, file_path):
        """Abre um arquivo com o programa padrão do sistema"""
        try:
            if sys.platform == 'win32':
                os.startfile(file_path)
            elif sys.platform == 'darwin':  # macOS
                subprocess.run(['open', file_path])
            else:  # Linux
                subprocess.run(['xdg-open', file_path])
        except Exception as e:
            logger.error(f"Erro ao abrir arquivo: {e}")
            messagebox.showerror("Erro", f"Não foi possível abrir o arquivo: {str(e)}")

    def comparative_analysis_dialog(self):
        """Abre diálogo para análise comparativa entre duas análises"""
        if not self.current_profile or self.current_profile not in self.profiles:
            messagebox.showinfo("Aviso", "Selecione um perfil para realizar análise comparativa")
            return
        
        profile = self.profiles[self.current_profile]
        
        if not profile.get('analyses') or len(profile['analyses']) < 2:
            messagebox.showinfo("Aviso", "São necessárias pelo menos duas análises para comparação")
            return
        
        dialog = tk.Toplevel(self.root)
        dialog.title(f"Análise Comparativa - {profile['name']}")
        dialog.geometry("700x500")
        dialog.resizable(True, True)
        dialog.transient(self.root)
        dialog.grab_set()
        
        # Container principal
        main_frame = ttk.Frame(dialog, padding=20)
        main_frame.pack(fill=tk.BOTH, expand=True)
        
        # Título
        ttk.Label(main_frame, text="Comparar Análises", font=("Arial", 14, "bold")).pack(pady=(0, 20))
        
        # Ordenar análises por data (mais recentes primeiro)
        sorted_analyses = sorted(profile['analyses'], key=lambda x: x['timestamp'], reverse=True)
        
        # Frame para seleção de análises
        selection_frame = ttk.Frame(main_frame)
        selection_frame.pack(fill=tk.X, pady=10)
        
        # Duas colunas para seleção
        left_column = ttk.LabelFrame(selection_frame, text="Primeira Análise")
        left_column.pack(side=tk.LEFT, fill=tk.BOTH, expand=True, padx=(0, 10))
        
        right_column = ttk.LabelFrame(selection_frame, text="Segunda Análise")
        right_column.pack(side=tk.LEFT, fill=tk.BOTH, expand=True)
        
        # Listas de seleção
        first_analysis_var = tk.StringVar()
        second_analysis_var = tk.StringVar()
        
        # Criar opções para combobox
        analysis_options = []
        analysis_ids = {}
        
        for analysis in sorted_analyses:
            timestamp = datetime.fromisoformat(analysis['timestamp']).strftime('%d/%m/%Y %H:%M')
            option_text = f"{timestamp}"
            analysis_options.append(option_text)
            analysis_ids[option_text] = analysis['id']
        
        # Comboboxes
        ttk.Combobox(left_column, textvariable=first_analysis_var, values=analysis_options, state="readonly").pack(fill=tk.X, padx=10, pady=10)
        ttk.Combobox(right_column, textvariable=second_analysis_var, values=analysis_options, state="readonly").pack(fill=tk.X, padx=10, pady=10)
        
        # Definir valores iniciais (duas análises mais recentes)
        if len(analysis_options) >= 2:
            first_analysis_var.set(analysis_options[0])
            second_analysis_var.set(analysis_options[1])
        
        # Frame para exibir comparação
        comparison_frame = ttk.LabelFrame(main_frame, text="Resultado da Comparação")
        comparison_frame.pack(fill=tk.BOTH, expand=True, pady=10)
        
        # Área de texto com scroll para mostrar resultados
        comparison_text = scrolledtext.ScrolledText(comparison_frame, wrap=tk.WORD, height=15)
        comparison_text.pack(fill=tk.BOTH, expand=True, padx=10, pady=10)
        comparison_text.insert(tk.END, "Selecione duas análises e clique em 'Comparar' para ver os resultados.")
        comparison_text.config(state=tk.DISABLED)
        
        # Botões
        buttons_frame = ttk.Frame(main_frame)
        buttons_frame.pack(fill=tk.X, pady=10)
        
        ttk.Button(buttons_frame, text="Fechar", command=dialog.destroy).pack(side=tk.RIGHT, padx=5)
        
        def compare_analyses():
            first_option = first_analysis_var.get()
            second_option = second_analysis_var.get()
            
            if not first_option or not second_option:
                messagebox.showinfo("Aviso", "Selecione duas análises para comparação")
                return
            
            if first_option == second_option:
                messagebox.showinfo("Aviso", "Selecione duas análises diferentes")
                return
            
            # Obter IDs das análises
            first_id = analysis_ids[first_option]
            second_id = analysis_ids[second_option]
            
            # Encontrar as análises correspondentes
            first_analysis = next((a for a in profile['analyses'] if a['id'] == first_id), None)
            second_analysis = next((a for a in profile['analyses'] if a['id'] == second_id), None)
            
            if not first_analysis or not second_analysis:
                messagebox.showerror("Erro", "Não foi possível encontrar as análises selecionadas")
                return
            
            # Gerar comparação
            comparison_result = self.generate_comparison(first_analysis, second_analysis)
            
            # Exibir resultado
            comparison_text.config(state=tk.NORMAL)
            comparison_text.delete(1.0, tk.END)
            comparison_text.insert(tk.END, comparison_result)
            comparison_text.config(state=tk.DISABLED)
        
        ttk.Button(buttons_frame, text="Comparar", command=compare_analyses).pack(side=tk.RIGHT, padx=5)
        
        def export_comparison():
            first_option = first_analysis_var.get()
            second_option = second_analysis_var.get()
            
            if not first_option or not second_option or first_option == second_option:
                messagebox.showinfo("Aviso", "Selecione duas análises diferentes para exportar")
                return
            
            # Obter resultado da comparação
            comparison_content = comparison_text.get(1.0, tk.END)
            
            if comparison_content.strip() == "Selecione duas análises e clique em 'Comparar' para ver os resultados.":
                messagebox.showinfo("Aviso", "Realize a comparação antes de exportar")
                return
            
            # Abrir diálogo para salvar
            save_path = filedialog.asksaveasfilename(
                title="Exportar Comparação",
                defaultextension=".txt",
                filetypes=[("Arquivo de texto", "*.txt"), ("Todos os arquivos", "*.*")]
            )
            
            if not save_path:
                return
            
            try:
                with open(save_path, 'w', encoding='utf-8') as f:
                    f.write(comparison_content)
                
                response = messagebox.askyesno(
                    "Comparação Exportada",
                    f"Comparação exportada com sucesso para:\n{save_path}\n\nDeseja abrir o arquivo agora?"
                )
                
                if response:
                    self.open_file(save_path)
                    
            except Exception as e:
                logger.error(f"Erro ao exportar comparação: {e}")
                messagebox.showerror("Erro", f"Não foi possível exportar a comparação: {str(e)}")
        
        ttk.Button(buttons_frame, text="Exportar", command=export_comparison).pack(side=tk.RIGHT, padx=5)
    
    def generate_comparison(self, first_analysis, second_analysis):
        """Gera uma comparação entre duas análises"""
        try:
            # Datas das análises
            first_date = datetime.fromisoformat(first_analysis['timestamp']).strftime('%d/%m/%Y %H:%M')
            second_date = datetime.fromisoformat(second_analysis['timestamp']).strftime('%d/%m/%Y %H:%M')
            
            # Cabeçalho
            result = f"COMPARAÇÃO DE ANÁLISES\n"
            result += f"{'='*60}\n\n"
            result += f"Análise 1: {first_date}\n"
            result += f"Análise 2: {second_date}\n\n"
            
            # Estatísticas básicas
            result += f"ESTATÍSTICAS BÁSICAS\n"
            result += f"{'-'*60}\n"
            
            # Número de mensagens
            first_msgs = first_analysis.get('conversation_meta', {}).get('message_count', 'N/A')
            second_msgs = second_analysis.get('conversation_meta', {}).get('message_count', 'N/A')
            
            result += f"Mensagens totais: {first_msgs} → {second_msgs}\n"
            
            # Mensagens do alvo
            first_target = first_analysis.get('conversation_meta', {}).get('target_message_count', 'N/A')
            second_target = second_analysis.get('conversation_meta', {}).get('target_message_count', 'N/A')
            
            result += f"Mensagens da pessoa: {first_target} → {second_target}\n"
            
            # Áudios
            first_audio = first_analysis.get('summary', {}).get('overview', {}).get('audio_messages', 'N/A')
            second_audio = second_analysis.get('summary', {}).get('overview', {}).get('audio_messages', 'N/A')
            
            if first_audio != 'N/A' and second_audio != 'N/A':
                result += f"Mensagens de áudio: {first_audio} → {second_audio}\n"
            
            # Tempo de resposta
            first_time = first_analysis.get('summary', {}).get('communication_patterns', {}).get('avg_response_time')
            second_time = second_analysis.get('summary', {}).get('communication_patterns', {}).get('avg_response_time')
            
            if first_time is not None and second_time is not None:
                # Formatação amigável
                def format_time(minutes):
                    if minutes < 60:
                        return f"{minutes:.1f} min"
                    else:
                        hours = minutes // 60
                        mins = minutes % 60
                        return f"{int(hours)}h {int(mins)}min"
                
                first_time_str = format_time(first_time)
                second_time_str = format_time(second_time)
                
                # Comparação
                time_diff = second_time - first_time
                if abs(time_diff) > 5:  # Diferença significativa (> 5 minutos)
                    if time_diff > 0:
                        time_trend = f"↑ (+{format_time(time_diff)})"
                    else:
                        time_trend = f"↓ ({format_time(time_diff)})"
                else:
                    time_trend = "→ (estável)"
                
                result += f"Tempo médio de resposta: {first_time_str} → {second_time_str} {time_trend}\n"
            
            # Comparação de atitudes
            result += f"\nATITUDES\n"
            result += f"{'-'*60}\n"
            
            first_attitudes = first_analysis.get('summary', {}).get('communication_patterns', {}).get('attitude_distribution', {})
            second_attitudes = second_analysis.get('summary', {}).get('communication_patterns', {}).get('attitude_distribution', {})
            
            # Todas as atitudes nas duas análises
            all_attitudes = set(list(first_attitudes.keys()) + list(second_attitudes.keys()))
            
            for attitude in sorted(all_attitudes):
                first_value = first_attitudes.get(attitude, 0) * 100
                second_value = second_attitudes.get(attitude, 0) * 100
                
                # Comparação
                diff = second_value - first_value
                if abs(diff) > 5:  # Diferença significativa (> 5%)
                    if diff > 0:
                        trend = f"↑ (+{diff:.1f}%)"
                    else:
                        trend = f"↓ ({diff:.1f}%)"
                else:
                    trend = "→"
                
                result += f"{attitude}: {first_value:.1f}% → {second_value:.1f}% {trend}\n"
            
            # Comparação de emoções
            first_emotions = first_analysis.get('summary', {}).get('content_analysis', {}).get('emotion_distribution', {})
            second_emotions = second_analysis.get('summary', {}).get('content_analysis', {}).get('emotion_distribution', {})
            
            if first_emotions and second_emotions:
                result += f"\nEMOÇÕES\n"
                result += f"{'-'*60}\n"
                
                # Todas as emoções nas duas análises
                all_emotions = set(list(first_emotions.keys()) + list(second_emotions.keys()))
                
                for emotion in sorted(all_emotions):
                    first_value = first_emotions.get(emotion, 0) * 100
                    second_value = second_emotions.get(emotion, 0) * 100
                    
                    # Comparação
                    diff = second_value - first_value
                    if abs(diff) > 5:  # Diferença significativa (> 5%)
                        if diff > 0:
                            trend = f"↑ (+{diff:.1f}%)"
                        else:
                            trend = f"↓ ({diff:.1f}%)"
                    else:
                        trend = "→"
                    
                    result += f"{emotion}: {first_value:.1f}% → {second_value:.1f}% {trend}\n"
            
            # Comparação de comportamentos
            first_behaviors = first_analysis.get('summary', {}).get('behavioral_analysis', {}).get('behavior_distribution', {})
            second_behaviors = second_analysis.get('summary', {}).get('behavioral_analysis', {}).get('behavior_distribution', {})
            
            if first_behaviors and second_behaviors:
                result += f"\nCOMPORTAMENTOS\n"
                result += f"{'-'*60}\n"
                
                # Tradução amigável dos comportamentos
                behavior_names = {
                    'avoidance': 'Evitação',
                    'approval_seeking': 'Busca de aprovação',
                    'control': 'Controle',
                    'emotional': 'Emocional',
                    'conflict': 'Conflito interno',
                    'proactive': 'Proativo',
                    'reactive': 'Reativo',
                    'status_focused': 'Focado em status',
                    'responsibility_shifting': 'Transferência de responsabilidade',
                    'neutral': 'Neutro'
                }
                
                # Todos os comportamentos nas duas análises
                all_behaviors = set(list(first_behaviors.keys()) + list(second_behaviors.keys()))
                
                for behavior in sorted(all_behaviors):
                    behavior_name = behavior_names.get(behavior, behavior)
                    first_value = first_behaviors.get(behavior, 0) * 100
                    second_value = second_behaviors.get(behavior, 0) * 100
                    
                    # Comparação
                    diff = second_value - first_value
                    if abs(diff) > 5:  # Diferença significativa (> 5%)
                        if diff > 0:
                            trend = f"↑ (+{diff:.1f}%)"
                        else:
                            trend = f"↓ ({diff:.1f}%)"
                    else:
                        trend = "→"
                    
                    result += f"{behavior_name}: {first_value:.1f}% → {second_value:.1f}% {trend}\n"
            
            # Comparação de indicadores psicológicos
            first_indicators = first_analysis.get('summary', {}).get('psychological_profile', {}).get('indicators', {})
            second_indicators = second_analysis.get('summary', {}).get('psychological_profile', {}).get('indicators', {})
            
            if first_indicators and second_indicators:
                result += f"\nINDICADORES PSICOLÓGICOS\n"
                result += f"{'-'*60}\n"
                
                # Tradução amigável dos indicadores
                indicator_names = {
                    'control_need': 'Necessidade de controle',
                    'approval_need': 'Necessidade de aprovação',
                    'uncertainty_tolerance': 'Tolerância a incertezas',
                    'emotional_openness': 'Abertura emocional',
                    'cognitive_flexibility': 'Flexibilidade cognitiva',
                    'autonomy': 'Autonomia',
                    'vulnerability': 'Vulnerabilidade'
                }
                
                # Todos os indicadores nas duas análises
                all_indicators = set(list(first_indicators.keys()) + list(second_indicators.keys()))
                
                for indicator in sorted(all_indicators):
                    indicator_name = indicator_names.get(indicator, indicator)
                    
                    first_score = first_indicators.get(indicator, {}).get('average_score', 'N/A')
                    second_score = second_indicators.get(indicator, {}).get('average_score', 'N/A')
                    
                    if first_score != 'N/A' and second_score != 'N/A':
                        # Comparação
                        diff = second_score - first_score
                        if abs(diff) > 0.5:  # Diferença significativa (> 0.5 pontos)
                            if diff > 0:
                                trend = f"↑ (+{diff:.1f})"
                            else:
                                trend = f"↓ ({diff:.1f})"
                        else:
                            trend = "→"
                        
                        result += f"{indicator_name}: {first_score:.1f}/10 → {second_score:.1f}/10 {trend}\n"
            
            # Comparação de temas
            first_themes = first_analysis.get('summary', {}).get('content_analysis', {}).get('main_themes', [])
            second_themes = second_analysis.get('summary', {}).get('content_analysis', {}).get('main_themes', [])
            
            if first_themes and second_themes:
                result += f"\nTEMAS DISCUTIDOS\n"
                result += f"{'-'*60}\n"
                
                # Converter para dicionários para facilitar comparação
                first_themes_dict = {theme: count for theme, count in first_themes}
                second_themes_dict = {theme: count for theme, count in second_themes}
                
                # Todos os temas nas duas análises
                all_themes = set(list(first_themes_dict.keys()) + list(second_themes_dict.keys()))
                
                for theme in sorted(all_themes):
                    theme_name = theme.replace('#', '')
                    first_count = first_themes_dict.get(theme, 0)
                    second_count = second_themes_dict.get(theme, 0)
                    
                    # Comparação
                    diff = second_count - first_count
                    if diff != 0:  # Qualquer diferença é significativa
                        if diff > 0:
                            trend = f"↑ (+{diff})"
                        else:
                            trend = f"↓ ({diff})"
                    else:
                        trend = "→"
                    
                    result += f"{theme_name}: {first_count} → {second_count} {trend}\n"
            
            # Análise de mudanças significativas
            result += f"\nMUDANÇAS SIGNIFICATIVAS\n"
            result += f"{'-'*60}\n"
            
            # Lista para armazenar mudanças significativas
            significant_changes = []
            
            # Verificar mudanças em atitudes
            for attitude in all_attitudes:
                first_value = first_attitudes.get(attitude, 0) * 100
                second_value = second_attitudes.get(attitude, 0) * 100
                diff = second_value - first_value
                
                if abs(diff) > 15:  # Mudança muito significativa (> 15%)
                    if diff > 0:
                        significant_changes.append(f"Aumento expressivo na atitude '{attitude}': +{diff:.1f}%")
                    else:
                        significant_changes.append(f"Redução expressiva na atitude '{attitude}': {diff:.1f}%")
            
            # Verificar mudanças em comportamentos
            if first_behaviors and second_behaviors:
                for behavior in all_behaviors:
                    behavior_name = behavior_names.get(behavior, behavior)
                    first_value = first_behaviors.get(behavior, 0) * 100
                    second_value = second_behaviors.get(behavior, 0) * 100
                    diff = second_value - first_value
                    
                    if abs(diff) > 15:  # Mudança muito significativa (> 15%)
                        if diff > 0:
                            significant_changes.append(f"Aumento expressivo no comportamento '{behavior_name}': +{diff:.1f}%")
                        else:
                            significant_changes.append(f"Redução expressiva no comportamento '{behavior_name}': {diff:.1f}%")
            
            # Verificar mudanças em indicadores psicológicos
            if first_indicators and second_indicators:
                for indicator in all_indicators:
                    indicator_name = indicator_names.get(indicator, indicator)
                    
                    first_score = first_indicators.get(indicator, {}).get('average_score', 'N/A')
                    second_score = second_indicators.get(indicator, {}).get('average_score', 'N/A')
                    
                    if first_score != 'N/A' and second_score != 'N/A':
                        diff = second_score - first_score
                        
                        if abs(diff) > 2:  # Mudança muito significativa (> 2 pontos)
                            if diff > 0:
                                significant_changes.append(f"Aumento expressivo em '{indicator_name}': +{diff:.1f} pontos")
                            else:
                                significant_changes.append(f"Redução expressiva em '{indicator_name}': {diff:.1f} pontos")
            
            # Verificar mudança no tempo de resposta
            if first_time is not None and second_time is not None:
                time_diff = second_time - first_time
                
                if abs(time_diff) > 30:  # Mudança significativa (> 30 minutos)
                    if time_diff > 0:
                        significant_changes.append(f"Aumento expressivo no tempo de resposta: +{time_diff:.1f} minutos")
                    else:
                        significant_changes.append(f"Redução expressiva no tempo de resposta: {time_diff:.1f} minutos")
            
            # Adicionar mudanças significativas ao resultado
            if significant_changes:
                for change in significant_changes:
                    result += f"• {change}\n"
            else:
                result += "Nenhuma mudança altamente significativa detectada entre as análises.\n"
            
            # Interpretação e recomendações
            result += f"\nINTERPRETAÇÃO E RECOMENDAÇÕES\n"
            result += f"{'-'*60}\n"
            
            # Gerar interpretações com base nas mudanças
            interpretations = []
            
            # Verificar padrão geral (melhoria ou piora na comunicação)
            improvement_indicators = 0
            degradation_indicators = 0
            
            # Considerar atitudes
            if 'colaborativa' in all_attitudes:
                diff = second_attitudes.get('colaborativa', 0) - first_attitudes.get('colaborativa', 0)
                if diff > 0.1:  # Aumento > 10%
                    improvement_indicators += 1
                elif diff < -0.1:  # Redução > 10%
                    degradation_indicators += 1
            
            if 'evasiva' in all_attitudes or 'resistente' in all_attitudes:
                evasive_diff = second_attitudes.get('evasiva', 0) - first_attitudes.get('evasiva', 0)
                resistant_diff = second_attitudes.get('resistente', 0) - first_attitudes.get('resistente', 0)
                
                # Redução em atitudes negativas é melhoria
                if evasive_diff < -0.1 or resistant_diff < -0.1:
                    improvement_indicators += 1
                # Aumento em atitudes negativas é degradação
                elif evasive_diff > 0.1 or resistant_diff > 0.1:
                    degradation_indicators += 1
            
            # Considerar tempo de resposta
            if first_time is not None and second_time is not None:
                time_ratio = second_time / first_time if first_time > 0 else 1
                
                if time_ratio < 0.8:  # Redução > 20%
                    improvement_indicators += 1
                elif time_ratio > 1.3:  # Aumento > 30%
                    degradation_indicators += 1
            
            # Gerar interpretação geral
            if improvement_indicators > degradation_indicators:
                interpretations.append("A comunicação apresenta sinais gerais de melhoria entre as duas análises.")
            elif degradation_indicators > improvement_indicators:
                interpretations.append("A comunicação apresenta alguns desafios crescentes entre as duas análises.")
            else:
                interpretations.append("A comunicação mostra padrões mistos de mudança, com aspectos que melhoraram e outros que apresentaram desafios.")
            
            # Interpretar mudanças específicas
            if significant_changes:
                for change in significant_changes[:3]:  # Até 3 mudanças mais significativas
                    if "Aumento expressivo na atitude 'colaborativa'" in change:
                        interpretations.append("Houve um aumento na disposição para colaboração, o que sugere que as abordagens utilizadas estão tendo efeito positivo.")
                    elif "Redução expressiva na atitude 'evasiva'" in change:
                        interpretations.append("A redução em comportamentos evasivos indica maior engajamento e disponibilidade para comunicação direta.")
                    elif "Aumento expressivo na atitude 'evasiva'" in change:
                        interpretations.append("O aumento em comportamentos evasivos pode indicar crescente desconforto ou hesitação em certos temas.")
                    elif "Aumento expressivo no comportamento 'Proativo'" in change:
                        interpretations.append("O aumento da proatividade sugere maior iniciativa e autonomia nas interações.")
                    elif "tempo de resposta" in change and "Redução" in change:
                        interpretations.append("A redução no tempo de resposta indica maior priorização da comunicação e possivelmente maior interesse nos assuntos discutidos.")
            
            # Adicionar interpretações ao resultado
            for interp in interpretations:
                result += f"• {interp}\n"
            
            # Adicionar recomendações específicas
            result += "\nRecomendações específicas com base nas mudanças:\n"
            
            # Gerar recomendações específicas
            recommendations = []
            
            # Baseadas em atitudes
            if 'evasiva' in all_attitudes and second_attitudes.get('evasiva', 0) > 0.25:  # > 25%
                recommendations.append("Continue usando perguntas diretas e solicite confirmações explícitas para reduzir a evasividade ainda presente.")
            
            if 'colaborativa' in all_attitudes and second_attitudes.get('colaborativa', 0) > first_attitudes.get('colaborativa', 0):
                recommendations.append("Reforce os comportamentos colaborativos usando reconhecimento explícito quando ocorrerem.")
            
            # Baseadas em comportamentos
            if first_behaviors and second_behaviors:
                if second_behaviors.get('avoidance', 0) > 0.2:  # > 20%
                    recommendations.append("Divida solicitações complexas em passos menores e mais gerenciáveis para reduzir a evitação.")
                
                if second_behaviors.get('approval_seeking', 0) > first_behaviors.get('approval_seeking', 0):
                    recommendations.append("Ofereça reconhecimento específico e constante para fortalecer a confiança e reduzir dependência de aprovação externa.")
            
            # Baseadas em indicadores psicológicos
            if first_indicators and second_indicators:
                if (second_indicators.get('uncertainty_tolerance', {}).get('average_score', 10) < 
                    first_indicators.get('uncertainty_tolerance', {}).get('average_score', 10)):
                    recommendations.append("Forneça estrutura e previsibilidade nas comunicações para acomodar a redução na tolerância a incertezas.")
                
                if (second_indicators.get('emotional_openness', {}).get('average_score', 0) > 
                    first_indicators.get('emotional_openness', {}).get('average_score', 0)):
                    recommendations.append("Aproveite a maior abertura emocional para construir conexão antes de abordar questões práticas.")
            
            # Recomendações genéricas se não houver suficientes
            if len(recommendations) < 3:
                recommendations.append("Continue monitorando padrões de comunicação para identificar tendências emergentes.")
                recommendations.append("Adapte as abordagens com base no contexto específico e estado emocional percebido.")
            
            # Adicionar recomendações ao resultado
            for rec in recommendations:
                result += f"• {rec}\n"
            
            return result
            
        except Exception as e:
            logger.error(f"Erro ao gerar comparação: {e}")
            return f"Erro ao gerar comparação: {str(e)}"

    def simulate_person_response(self, message, context, approach):
        """Simula uma resposta da pessoa-alvo com base em seu perfil comportamental"""
        try:
            profile = self.profiles[self.current_profile]
            
            # Obter padrões comportamentais do perfil
            behavioral_patterns = profile.get('behavioral_patterns', {})
            communication_style = behavioral_patterns.get('communication_style', {})
            
            # Obter estatísticas de comunicação
            communication_stats = profile.get('communication_stats', {})
            attitude_dist = communication_stats.get('attitude_distribution', {})
            
            # Obter perfil psicológico
            psych_profile = profile.get('psychological_profile', {})
            
            # Determinar atitude provável com base na distribuição histórica
            if attitude_dist:
                # Ajustar probabilidades com base no contexto e abordagem
                adjusted_dist = dict(attitude_dist)
                
                # Ajustes baseados no contexto/tema
                if context == "#plantao":
                    adjusted_dist['evasiva'] = adjusted_dist.get('evasiva', 0) * 1.2
                    adjusted_dist['resistente'] = adjusted_dist.get('resistente', 0) * 1.1
                elif context == "#feriado":
                    adjusted_dist['resistente'] = adjusted_dist.get('resistente', 0) * 1.3
                    adjusted_dist['evasiva'] = adjusted_dist.get('evasiva', 0) * 1.2
                elif context == "#pagamento":
                    adjusted_dist['colaborativa'] = adjusted_dist.get('colaborativa', 0) * 1.2
                elif context == "#reuniao":
                    adjusted_dist['neutra'] = adjusted_dist.get('neutra', 0) * 1.2
                elif context == "#reconhecimento":
                    adjusted_dist['colaborativa'] = adjusted_dist.get('colaborativa', 0) * 1.5
                    
                # Ajustes baseados na abordagem
                if approach == "elogio":
                    adjusted_dist['colaborativa'] = adjusted_dist.get('colaborativa', 0) * 1.5
                    adjusted_dist['resistente'] = adjusted_dist.get('resistente', 0) * 0.7
                elif approach == "direto":
                    adjusted_dist['resistente'] = adjusted_dist.get('resistente', 0) * 1.2
                    adjusted_dist['evasiva'] = adjusted_dist.get('evasiva', 0) * 1.1
                elif approach == "urgente":
                    adjusted_dist['colaborativa'] = adjusted_dist.get('colaborativa', 0) * 1.2
                    adjusted_dist['evasiva'] = adjusted_dist.get('evasiva', 0) * 0.8
                elif approach == "opcoes":
                    adjusted_dist['colaborativa'] = adjusted_dist.get('colaborativa', 0) * 1.3
                    adjusted_dist['evasiva'] = adjusted_dist.get('evasiva', 0) * 0.7
                
                # Normalizar distribuição ajustada
                total = sum(adjusted_dist.values())
                if total > 0:
                    adjusted_dist = {k: v / total for k, v in adjusted_dist.items()}
                
                # Determinar atitude mais provável
                probable_attitude = max(adjusted_dist.items(), key=lambda x: x[1])[0]
            else:
                # Padrão se não houver dados históricos
                probable_attitude = "neutra"
            
            # Determinar comportamento provável
            if communication_style:
                # Ajustar probabilidades com base no contexto e abordagem
                adjusted_behaviors = dict(communication_style)
                
                # Ajustes baseados no contexto/tema
                if context == "#problema":
                    adjusted_behaviors['avoidance'] = adjusted_behaviors.get('avoidance', 0) * 1.3
                    adjusted_behaviors['responsibility_shifting'] = adjusted_behaviors.get('responsibility_shifting', 0) * 1.4
                elif context == "#reconhecimento":
                    adjusted_behaviors['approval_seeking'] = adjusted_behaviors.get('approval_seeking', 0) * 1.3
                
                # Ajustes baseados na abordagem
                if approach == "elogio":
                    adjusted_behaviors['approval_seeking'] = adjusted_behaviors.get('approval_seeking', 0) * 1.2
                elif approach == "direto":
                    adjusted_behaviors['control'] = adjusted_behaviors.get('control', 0) * 1.2
                    
                # Normalizar distribuição ajustada
                total = sum(adjusted_behaviors.values())
                if total > 0:
                    adjusted_behaviors = {k: v / total for k, v in adjusted_behaviors.items()}
                
                # Determinar comportamento mais provável
                probable_behavior = max(adjusted_behaviors.items(), key=lambda x: x[1])[0]
            else:
                # Padrão se não houver dados históricos
                probable_behavior = "neutral"
            
            # Gerar resposta simulada com base na atitude e comportamento prováveis
            response = self.generate_simulated_response(message, probable_attitude, probable_behavior, context, approach, psych_profile)
            
            # Adicionar metadados à simulação
            simulation = f"{response}\n\n---\n"
            simulation += f"Atitude Provável: {probable_attitude}\n"
            simulation += f"Comportamento Provável: {probable_behavior}\n"
            
            # Probabilidade de resposta
            response_probability = self.calculate_response_probability(message, context, approach, profile)
            simulation += f"Probabilidade de resposta: {response_probability:.0f}%\n"
            
            # Tempo estimado de resposta
            response_time = communication_stats.get('avg_response_time')
            if response_time:
                # Ajustar com base na abordagem
                if approach == "urgente":
                    response_time *= 0.7
                elif approach == "elogio":
                    response_time *= 0.9
                
                # Formatar tempo
                if response_time < 60:
                    time_str = f"{response_time:.0f} minutos"
                else:
                    hours = int(response_time // 60)
                    mins = int(response_time % 60)
                    time_str = f"{hours}h{mins:02d}min"
                
                simulation += f"Tempo estimado de resposta: {time_str}\n"
            
            # Probabilidade de áudio vs. texto
            if 'content_type_distribution' in profile.get('communication_stats', {}):
                audio_prob = profile['communication_stats']['content_type_distribution'].get('audio', 0)
                text_prob = profile['communication_stats']['content_type_distribution'].get('text', 0)
                
                total = audio_prob + text_prob
                if total > 0:
                    audio_percentage = (audio_prob / total) * 100
                    simulation += f"Probabilidade de resposta por áudio: {audio_percentage:.0f}%\n"
            
            return simulation
            
        except Exception as e:
            logger.error(f"Erro ao simular resposta: {e}")
            return f"Erro ao gerar simulação: {str(e)}"

    def calculate_response_probability(self, message, context, approach, profile):
        """Calcula a probabilidade de resposta com base nos parâmetros"""
        # Probabilidade base
        base_probability = 80  # 80% chance de resposta por padrão
        
        # Ajustar com base no comprimento da mensagem
        message_length = len(message.split())
        if message_length > 100:  # Mensagem muito longa
            base_probability -= 15
        elif message_length < 10:  # Mensagem muito curta
            base_probability += 5
        
        # Ajustar com base no contexto
        context_adjustments = {
            "#plantao": 0,      # Neutro
            "#feriado": -10,    # Menos provável
            "#pagamento": +5,   # Mais provável
            "#reuniao": -5,     # Ligeiramente menos provável
            "#documentacao": 0, # Neutro
            "#problema": -15,   # Muito menos provável
            "#reconhecimento": +15  # Muito mais provável
        }
        
        if context in context_adjustments:
            base_probability += context_adjustments[context]
        
        # Ajustar com base na abordagem
        approach_adjustments = {
            "neutro": 0,        # Neutro
            "direto": -5,       # Ligeiramente menos provável
            "elogio": +15,      # Muito mais provável
            "urgente": +10,     # Mais provável
            "opcoes": +5        # Ligeiramente mais provável
        }
        
        base_probability += approach_adjustments.get(approach, 0)
        
        # Ajustar com base no perfil psicológico, se disponível
        if 'psychological_profile' in profile:
            psych = profile['psychological_profile']
            
            # Necessidade de aprovação aumenta resposta a elogios
            if ('motivation_factors' in psych and 
                'approval_need' in psych['motivation_factors'] and 
                approach == "elogio"):
                
                approval_need = psych['motivation_factors']['approval_need'].get('score', 5)
                base_probability += (approval_need - 5) * 3  # -15 a +15
            
            # Necessidade de controle reduz resposta a abordagens diretas
            if ('motivation_factors' in psych and 
                'control_need' in psych['motivation_factors'] and 
                approach == "direto"):
                
                control_need = psych['motivation_factors']['control_need'].get('score', 5)
                base_probability -= (control_need - 5) * 2  # -10 a +10
        
        # Garantir que a probabilidade fique entre 0% e 100%
        return max(0, min(100, base_probability))

                self.open_file(save_path)

            messagebox.showerror("Erro", f"Não foi possível exportar o relatório: {str(e)}")
